namespace = birth

#Sickly child recovery - for child
birth.3034 = {
	type = character_event
	title = birth.3034.t
	desc = birth.3034.desc
	theme = physical_health
	left_portrait = {
		character = root
		animation = happiness
	}
	#ADDED 
	id_override_priority = 228

	immediate = {
		play_music_cue = mx_cue_positive_effect
		save_scope_as = child
		set_favorite_toy_effect = yes
		if = {
			limit = {
				exists = mother
			}
			mother = {
				trigger_event = birth.3035
			}
		}
		if = {
			limit = {
				exists = father
			}
			father = {
				trigger_event = birth.3035
			}
		}
	}

	option = {
		name = birth.3034.a
		remove_trait = sickly
		#ADDED
		if = {
			limit = { NOT = { has_trait = physique_good } }
			random = {
				chance = 25
				add_trait_force_tooltip = weak
			}
		}
	}
}
