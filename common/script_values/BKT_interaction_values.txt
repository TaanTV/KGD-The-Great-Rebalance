normal_ransom_cost_value = {
	value = ransom_cost
	# Some cultures are better at ransoming than others.
	if = {
		limit = {
			exists = imprisoner
			imprisoner.culture = { has_cultural_parameter = can_demand_higher_ransoms_from_lower_tiers }
			# Kings and emperors are already worth more, so we just boost landless and ducal-tier or below.
			OR = {
				NOT = { exists = liege }
				highest_held_title_tier <= tier_duchy
			}
		}
		multiply = 1.5
	}
	# Rich guests can pay a bit more...
	if = {
		limit = {
			OR = {
				NOT = { exists = liege }
				scope:recipient ?= this  # when used in golden_obligation_value etc. they are doing on their own behavior so we don't care liege
			}
			is_ruler = no
		}
		if = {
			limit = { gold >= 100 }
			add = 90
		}
		else_if = {
			limit = { gold >= 50 }
			add = 15
		}
	}
	if = {
		limit = {
			has_game_rule = hard_difficulty
			OR = {
				scope:imprisoner ?= { is_ai = no }
				imprisoner ?= { is_ai = no }
			}
		}
		multiply = 0.75		#prev 0.5
	}
	else_if = {
		limit = {
			has_game_rule = very_hard_difficulty
			OR = {
				scope:imprisoner ?= { is_ai = no }
				imprisoner ?= { is_ai = no }
			}
		}
		multiply = 0.5		#prev 0.25
	}
	
	#ADDED - scale with diplo difference
	if = {
		limit = {
			exists = scope:imprisoner
			exists = scope:payer
		}
		multiply = {
			value = 1
			add = {
				value = scope:imprisoner.diplomacy
				add = 2		#offsets the values a bit so 1 vs 2 diplo isnt an x2 difference
				divide = { value = scope:payer.diplomacy add = 2 }
				min = 0.5
				max = 2			#range 0.5  to 2
				add = -1.5	 	#range -1   to 0.5
				multiply = 0.4	#range -0.4 to 0.2
								#final range 0.6 to 1.2
			}
		}
		round_to = 5
	}
	else_if = {
		limit = {
			exists = imprisoner
			#that means THIS is the imprisoned
		}
		multiply = {
			value = 1
			add = {
				value = imprisoner.diplomacy
				add = 2		#offsets the values a bit so 1 vs 2 diplo isnt an x2 difference
				if = {
					limit = {
						exists = scope:puppet_or_actor
						exists = scope:secondary_recipient
						NOT = { scope:puppet_or_actor = scope:secondary_recipient.imprisoner }
					}
					#We're getting ransomed by actor, good!
					divide = { value = scope:puppet_or_actor.diplomacy add = 2 }
				}
				else_if = {
					limit = {
						exists = scope:recipient
						exists = scope:secondary_recipient
						NOT = { scope:recipient = scope:secondary_recipient.imprisoner }
					}
					#I guess we're getting ransomed by recipient instead, awesome!
					divide = { value = scope:recipient.diplomacy add = 2 }
				}
				else = {	
					#Just in case.
					divide = { value = diplomacy add = 2 }
				}
				min = 0.5
				max = 2			#range 0.5  to 2
				add = -1.5	 	#range -1   to 0.5
				multiply = 0.4	#range -0.4 to 0.2
								#final range 0.6 to 1.2
			}
		}
		round_to = 5
	}
	
	multiply = haggler_ransom_cost_factor
}

marriage_female_fertility_cutoff_age_value = {
	value = {
		add = define:NChildbirth|MAX_FEMALE_REPRODUCTION_AGE
		add = -8		#prev -3
		if = {
			limit = {
				has_trait = fecund
			}
			add = 5
		}
		if = {
			limit = {
				dynasty ?= {
					has_dynasty_perk = blood_legacy_5
				}
			}
			add = 5
		}
	}
}

offer_vassalage_acceptance_value = {
	value = {
		if = {
			limit = {
				exists = scope:actor
				exists = scope:recipient
			}
			#ADDED - rewrote this completely
			add = {
				value = -40		#at 0%
				multiply = {
					value = 100
					scope:actor.culture = { subtract = "cultural_acceptance(scope:recipient.culture)" }
					divide = 100
				}
			}
			if = {
				limit = { scope:actor = { knows_language_of_culture = scope:recipient.culture } }
				multiply = 0.5
			}
		}
		else = {
			#If the scopes are not there somehow...
			add = -20
		}
	}
}
