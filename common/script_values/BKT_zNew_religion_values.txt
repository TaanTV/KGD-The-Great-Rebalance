
#Is used in rite_creation_cost_mult:
# Modify (multiply) the doctrine cost when creating a faith.
# The returned value is interpreted as a multiplicative modifier (e.g. 0.5 means +50% cost).
# scope:character = the creation character
# scope:old_rite = the rite the character had before creating
# scope:old_faith = the faith the character is converting from
modify_rite_cost_multiplier = {
	add = {
		value = 1
		desc = RITE_CREATION_VIEW_EDIT
		format = "BREAKDOWN_FORMAT_PIETY_PERCENTAGE_POSITIVE_IS_BAD"
	}
	if = {
		limit = { 
			scope:old_faith ?= {
				religious_head ?= scope:character
				#religious_head_title.tier >= tier_kingdom
			}
		}
		add = {
			value = 1
			desc = pam_kingdom_of_heaven_decision_hof_tt
			format = "BREAKDOWN_FORMAT_PIETY_PERCENTAGE_POSITIVE_IS_BAD"
		}
	}
	if = {
		limit = { 
			scope:old_rite ?= {
				rite_has_parameter = pluralism_fundamentalist_much_more_expensive_to_modify_rite
			}
		}
		add = {
			value = 1
			desc = doctrine_pluralism_fundamentalist_name
			format = "BREAKDOWN_FORMAT_PIETY_PERCENTAGE_POSITIVE_IS_BAD"
		}
	}
	else_if = {
		limit = { 
			scope:old_rite ?= {
				rite_has_parameter = pluralism_righteous_more_expensive_to_modify_rite
			}
		}
		add = {
			value = 0.5
			desc = doctrine_pluralism_righteous_name
			format = "BREAKDOWN_FORMAT_PIETY_PERCENTAGE_POSITIVE_IS_BAD"
		}
	}
}
