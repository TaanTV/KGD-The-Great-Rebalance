# Provisions loss from courtiers

laamp_courtier_provisions_loss_positive = {
	value = 0
	every_courtier_or_guest = {
		add = 3
		# Inappetetic people obviously eat a lot less
		if = {
			limit = { has_trait = inappetetic }
			add = -2
		}
		# Monks eat less because idk that sounds logical
		else_if = {
			limit = { has_trait = devoted }
			add = -1
		}
		# Gluttonous and comfort eaters eat more.
		if = {
			limit = { OR = { has_trait = comfort_eater has_trait = gluttonous } }
			add = 2
		}
		if = {
			limit = { prowess > 12 }
			add = {
				value = prowess
				subtract = 12
				divide = 6
				floor = yes
			}
		}
		floor = yes
		min = 0
	}
	if = {
		limit = { has_game_rule = kan_laamp_courtier_provision_maintaince_halved }
		multiply = 0.5
		ceiling = yes
	}
	else_if = {
		limit = { has_game_rule = kan_laamp_courtier_provision_maintaince_doubled }
		multiply = 2
	}
}

		value = scope:employer.yearly_character_income
		if = {
			limit = {
				scope:employer.gold > 0
			}

		}
	}
