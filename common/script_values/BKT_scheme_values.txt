# Secrecy Values

# Cap secrecy for the scheme_owner to a sensible maximum value so that you can't single-handedly cover up for every gossip in the world.
secrecy_skill_contribution_max_value = 45		#prev 40

secrecy_base_value = {
	add = {
		value = 30		#prev 20
		desc = BASE
	}
	# Reduce by the target's scheme discovery chance, if appropriate.
	if = {
		limit = {
			# Scheme has to exist already.
			exists = this
			#ADDED - prevent startup errors
			exists = scope:owner
			# We gotta be facing a character.
			is_scheme_target_type = character
			# Do they have a spymaster...
			scheme_target_character.cp:councillor_spymaster ?= {
				# ... who is performing the appropriate task...
				is_performing_council_task = task_disrupt_schemes
				NOR = {
					# ... who _isn't_ the scheme owner...
					this = scope:owner
					# ... or an agent in the scheme.
					save_temporary_scope_as = char_temp
					root = {
						any_scheme_agent_character = { this = scope:char_temp }
					}
				}
			}
		}
		scope:target = { save_temporary_scope_as = councillor_liege }
		add = {
			value = scope:target.cp:councillor_spymaster.spymaster_disrupt_schemes_total_modifier
			multiply = -1
			desc = secrecy.scheme_owner.spymaster.disrupt_schemes
		}
	}
	if = {
		limit = {
			OR = {
				scheme_target_character.host ?= scope:owner
				scheme_target_title.holder ?= scope:owner
			}
		}
		add = {
			value = 20
			desc = secrecy.own_court
		}
	}
	else_if = {
		limit = {
			# Scheme has to exist already.
			exists = this
			#ADDED - prevent startup errors
			exists = scope:owner
			# We gotta be facing a character.
			is_scheme_target_type = title
			# Do they have a spymaster...
			scheme_target_title.holder.cp:councillor_spymaster ?= {
				# ... who is performing the appropriate task...
				is_performing_council_task = task_disrupt_schemes
				NOR = {
					# ... who _isn't_ the scheme owner...
					this = scope:owner
					# ... or an agent in the scheme.
					save_temporary_scope_as = char_temp
					root = {
						any_scheme_agent_character = { this = scope:char_temp }
					}
				}
			}
		}
		add = {
			value = scope:target.holder.cp:councillor_spymaster.spymaster_disrupt_schemes_total_modifier
			multiply = -1
			desc = secrecy.scheme_owner.spymaster.disrupt_schemes
		}
	}
	#ADDED
	if = {
		limit = {
			always = no	#disable for now
			exists = this
			is_type_basic = no
			scope:owner ?= { is_ai = no }	#dunno about that 
		}
		add = secrecy_agent_count_reduction_value
	}
}
secrecy_agent_count_reduction_value = {
	add = {
		value = {
			every_scheme_agent_slot = {
				limit = {
					is_filled = yes
					NOT = { agent_slot_has_contribution_type = secrecy }
				}
				add = -3			#prev -10
			}
		}
		desc = secrecy.num_agents
	}
}

ai_will_do_redouble_guards_value = {
	if = {
		limit = {
			# We don't care about dpeloyed countermeasures here: guard against murders even if there's a seduction already happening.
			any_targeting_scheme = {
				is_scheme_exposed = yes
				var:apply_countermeasures ?= flag:calculating
			}
		}
		add = countermeasure_ai_will_do_prio2		#prev countermeasure_ai_will_do_prio4
	}
	else = { add = countermeasure_ai_will_do_prio0 }
}
ai_will_do_withdraw_from_view_value = {
	if = {
		limit = {
			any_targeting_scheme = {
				is_scheme_exposed = yes
				OR = {
					var:apply_countermeasures ?= flag:calculating
					var:apply_countermeasures ?= flag:opportunistic
				}
				#ADDED - nothing else is worth secluding over it imo
				OR = {
					scheme_type = murder
					scheme_type = abduct
				}
			}
			#OR = {
			#	has_trait = humble
			#	has_trait = shy
			#	has_trait = eccentric
			#}
			NOR = {
				has_trait = arrogant
				has_trait = gregarious
				has_trait = stubborn
				has_trait = trusting
			}
		}
		#ADDED - this IF and moved original thing into an ELSE
		if = {
			limit = {
				OR = {
					has_trait = humble
					has_trait = shy
					has_trait = eccentric
				}
			}
			add = countermeasure_ai_will_do_prio4
		}
		else = {
		add = countermeasure_ai_will_do_prio3		#prev countermeasure_ai_will_do_prio1
		}
	}
	else = { add = countermeasure_ai_will_do_prio0 }
