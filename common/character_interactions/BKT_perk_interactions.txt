
claim_throne_interaction = {
	icon = icon_scheme_claim_throne
	category = interaction_category_vassal
	interface_priority = 268 # after Start Claimant Faction

	ai_targets = {
		ai_recipients = liege
	}

	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes


	scheme = claim_throne

	desc = claim_throne_interaction_desc

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			learning >= medium_skill_rating
			intrigue >= mediocre_skill_rating
			ai_greed >= medium_positive_ai_value
			is_adult = yes
			NOR = {
				has_trait = content
				opinion = {
					target = liege
					value >= high_positive_opinion
				}
				any_scheme = {
					OR = {
						scheme_type = murder
						scheme_type = abduct
						scheme_type = claim_throne
						scheme_type = fabricate_hook
						scheme_type = steal_back_artifact
					}
				}
			}
		}
	}

	is_shown = {
		scope:puppet_or_actor ?= {
			highest_held_title_tier > tier_barony
			is_landed = yes
			is_physically_able_adult = yes
			liege ?= scope:recipient
			has_access_to_claim_throne_scheme_trigger = { TARGET = scope:recipient }
		}
		scope:recipient != scope:puppet_or_actor
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor = {
			is_physically_able_adult = yes
			can_start_scheme = {
				target_character = scope:recipient
				type = claim_throne
			}
		}
	}

	is_highlighted = {
		scope:puppet_or_actor = {
			is_diarch_of_target = scope:recipient
		}
		scope:recipient = {
			#NOT = { has_diarchy_parameter = can_be_instantly_dismissed }
			#ADDED - i think it's supposed to be this?
			NOT = { has_diarchy_active_parameter = can_be_instantly_dismissed }
		}
	}

	highlighted_reason = HIGHLIGHTED_IS_DIARCH_REASON

	# Scheme Starter Packages
	options_heading = schemes.t.agent_packages
	send_options_exclusive = yes
	## Balanced agents.
	send_option = {
		flag = agent_focus_balance
		current_description = claim_throne_interaction.tt.agent_focus_balance
	}
	## Focused on Success Chance.
	send_option = {
		flag = agent_focus_success
		current_description = claim_throne_interaction.tt.agent_focus_success
	}
	## Focused on Speed.
	send_option = {
		flag = agent_focus_speed
		current_description = claim_throne_interaction.tt.agent_focus_speed
	}
	## Focused on Secrecy.
	send_option = {
		flag = agent_focus_secrecy
		current_description = claim_throne_interaction.tt.agent_focus_secrecy
	}
	on_send = {
		scope:puppet_or_actor = {
			if = {
				limit = {
					OR = {
						has_character_modifier = pam_granted_indulgences_modifier
						has_character_modifier = pam_granted_indulgences_modifier_small
					}
				}
				change_spiritual_fulfillment = medium_spiritual_fulfillment_loss
			}
		}
	}

	on_accept = {
		scope:puppet_or_actor = {
			stress_and_fulfillment_impact = {
				content = medium_stress_impact_gain
				just = medium_stress_impact_gain
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = claim_throne_interaction_toast
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient

				# Success.
				if = {
					limit = { scope:agent_focus_success ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = claim_throne
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Success.
						AGENT_1 = agent_diplomat
						AGENT_2 = agent_diplomat
						AGENT_3 = agent_scribe
						# Speed.
						AGENT_4 = agent_cleric
						# Secrecy.
						AGENT_5 = agent_justiciar
					}
				}
				# Speed.
				else_if = {
					limit = { scope:agent_focus_speed ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = claim_throne
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Speed.
						AGENT_1 = agent_cleric
						AGENT_2 = agent_socialite
						AGENT_3 = agent_gabbler
						# Success.
						AGENT_4 = agent_diplomat
						# Secrecy.
						AGENT_5 = agent_justiciar
					}
				}
				# Secrecy.
				else_if = {
					limit = { scope:agent_focus_secrecy ?= yes }
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = claim_throne
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Secrecy.
						AGENT_1 = agent_justiciar
						AGENT_2 = agent_justiciar
						AGENT_3 = agent_decoy
						# Success.
						AGENT_4 = agent_diplomat
						# Speed.
						AGENT_5 = agent_cleric
					}
				}
				# Balanced.
				else = {
					begin_scheme_with_agents_effect = {
						SCHEME_TYPE = claim_throne
						TARGET_TYPE = target_character
						TARGET_SCOPE = scope:recipient
						# Success.
						AGENT_1 = agent_diplomat
						AGENT_2 = agent_scribe
						# Speed.
						AGENT_3 = agent_cleric
						AGENT_4 = agent_socialite
						# Secrecy.
						AGENT_5 = agent_decoy
					}
				}

				show_as_tooltip = {
					stress_and_fulfillment_impact = {
						content = medium_stress_impact_gain
						just = medium_stress_impact_gain
					}
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = claim_throne_interaction_toast_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = minor_unity_loss
			DESC = clan_unity_throne_claim.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}

	ai_frequency_by_tier = {
		barony = 0
		county = 48
		duchy = 36
		kingdom = 36
		empire = 0
		hegemony = 0
	}

	ai_will_do = {
		base = 100

		# Agenda impact
		modifier = {
			scope:puppet_or_actor = {
				any_character_struggle = {
					involvement = involved
					is_struggle_type = iberian_struggle
				}
			}
			add = {
				value = 0
				if = {
					limit = {
						scope:puppet_or_actor = {
							any_character_struggle = {
								phase_has_catalyst = catalyst_gain_claim_on_title_fp2
							}
							has_character_flag = agenda_towards_escalation
						}
					}
					add = 200
				}
				else_if = {
					limit = {
						scope:puppet_or_actor = {
							any_character_struggle = {
								phase_has_catalyst = catalyst_gain_claim_on_title_fp2
							}
						}
					}
					add = -100
				}
			}
		}

		modifier = {
			scope:puppet_or_actor = {
				any_character_struggle = {
					involvement = involved
					is_struggle_type = persian_struggle
				}
			}
			add = {
				value = 0
				if = {
					limit = {
						scope:puppet_or_actor = {
							any_character_struggle = {
								phase_has_catalyst = catalyst_gain_claim_on_title_fp3
							}
							has_trait = fp3_struggle_detractor
						}
					}
					add = 200
				}
				else_if = {
					limit = {
						scope:puppet_or_actor = {
							any_character_struggle = {
								phase_has_catalyst = catalyst_gain_claim_on_title_fp3
							}
							has_trait = fp3_struggle_supporter
						}
						scope:recipient = {
							has_trait = fp3_struggle_detractor
						}
					}
					add = 100
				}
				else_if = {
					limit = {
						scope:puppet_or_actor = {
							any_character_struggle = {
								phase_has_catalyst = catalyst_gain_claim_on_title_fp3
							}
						}
					}
					add = -100
				}
			}
		}

		# Unity modifiers
		evaluate_action_decreasing_house_unity = {
			VALUE = 100
		}

		start_hostile_scheme_ai_base_modifiers = yes # At the end so Cowed can block it completely
	}
}

demand_payment_interaction = {
	category = interaction_category_friendly
	icon = icon_gold

	auto_accept = yes
	use_diplomatic_range = yes
	interface_priority = 35

	desc = demand_payment_interaction_desc

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			OR = {
				ai_greed >= low_positive_ai_value
				is_at_war = yes
			}
			NOT = {
				has_trait = generous
			}
		}
	}

	is_shown = {
		scope:puppet_or_actor ?= {
			OR = {
				has_perk = golden_obligations_perk
				AND = {
					any_character_struggle = {
						involvement = involved
						has_struggle_phase_parameter = unlocks_demand_payments_for_all
						is_secondary_character_involvement_either_trigger = {
							CHAR = scope:recipient
						}
					}
				}
			}
		}
		scope:recipient = {
			is_ai = yes
		}
		scope:recipient != scope:puppet_or_actor
	}

	is_valid_showing_failures_only = {
		scope:puppet_or_actor = {
			is_physically_able_adult = yes
			NOT = {
				is_at_war_with = scope:recipient
			}
		}
		scope:recipient = {
			is_adult = yes
			is_imprisoned = no
		}
		scope:recipient = {
			custom_description = {
				text = demand_payment_too_little_gold
				gold >= 1
			}
		}
	}

	is_valid = {
		scope:puppet_or_actor = {
			has_usable_hook = scope:recipient
			#ADDED
			has_strong_hook = scope:recipient
		}
	}

	is_highlighted = {
		scope:puppet_or_actor = {
			has_usable_hook = scope:recipient
			#ADDED
			has_strong_hook = scope:recipient
		}
	}

	on_accept = {
		if = {
			limit = {
				scope:recipient.gold < scope:recipient.golden_obligation_value
			}
			custom_tooltip = demand_payment_too_little_gold_tt
		}
		scope:puppet_or_actor = {
			send_interface_toast = {
				title = demand_payment_interaction_message
				left_icon = scope:recipient

				scope:puppet_or_actor = {
					use_hook = scope:recipient
				}
				scope:recipient = {
					pay_short_term_gold = {
						target = scope:puppet_or_actor
						gold = golden_obligation_value_check_current_situation
					}
					# ADDED. People probably dont like this shit.
					if = {
						limit = {
							NOT = { has_relation_grudge = scope:actor }
							"scope:recipient.opinion(scope:actor)" < -50
						}
						set_random_rivalry_reason = {
							TARGET = scope:actor
							RELATION = grudge
						}
					}
					else = {
						add_opinion = {
							target = scope:actor
							modifier = hate_opinion
							opinion = -30
						}
					}
				}
			}
			stress_and_fulfillment_impact = {
				generous = medium_stress_impact_gain
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					title = demand_payment_interaction_message_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}


		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = minor_unity_loss
			DESC = clan_unity_demand_payment.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}


	ai_target_quick_trigger = {
		adult = yes
	}

	ai_targets = {
		ai_recipients = hooked_characters
	}

	ai_frequency_by_tier = {
		barony = 0
		county = 48
		duchy = 36
		kingdom = 12
		empire = 6
		hegemony = 6
	}

	ai_will_do = {
		base = 100

		# Unity modifiers
		evaluate_action_increasing_house_unity = {
			VALUE = -100
		}

		modifier = { #ai won't use the interaction if they'll get nothing out of it
			factor = 0
			scope:recipient.gold < 8
		}

		modifier = {
			factor = 0
			scope:puppet_or_actor = {
				OR = {
					has_relation_friend = scope:recipient
					has_relation_lover = scope:recipient
					has_relation_best_friend = scope:recipient
					has_relation_soulmate = scope:recipient
				}
			}
		}
	}
}

buy_claim_interaction = {
	category = interaction_category_diplomacy
	icon = vassal_claim_liege_title_interaction

	desc = buy_claim_interaction_desc
	target_type = title
	target_filter = recipient_domain_titles
	ignores_pending_interaction_block = yes

	is_shown = {
		scope:puppet_or_actor ?= {
			is_landless_adventurer = no
			OR = {
				AND = {
					has_perk = sanctioned_loopholes_perk
					NOR = {
						government_has_flag = government_has_merit
						#Beseech claim interaction improved instead
						pam_beseech_claims_access_trigger = yes
						government_has_flag = government_is_japan_feudal
						government_has_flag = government_is_japan_administrative
					}
				}
				AND = {
					any_character_struggle = {
						involvement = involved
						has_struggle_phase_parameter = unlocks_buy_claim_for_all
						is_secondary_character_involvement_either_trigger = {
							CHAR = scope:recipient
						}
					}
				}
			}
			NOT = { target_is_liege_or_above = scope:recipient }
		}
		NOT = {
			scope:puppet_or_actor ?= scope:recipient
			AND = {
				scope:recipient = scope:puppet_or_actor.faith.religious_head
				scope:puppet_or_actor.rite ?= {
					NOT = { rite_has_doctrine = doctrine_theocracy_temporal }
				}
			}
		}
		scope:recipient = {
			is_playable_character = yes
			primary_title.tier > tier_barony
			NOT = {
				target_is_liege_or_above = scope:puppet_or_actor
			}
		}
	}

	can_be_picked_title = {
		scope:target = {
			tier >= tier_county
			#ADDED - "cannot_grant_ceremonial_title"? Close enough...
			custom_tooltip = {
				text = cannot_grant_ceremonial_title
				NOT = { has_variable = ceremonial_title }
			}
		}
	}
	#ADDED
	cooldown = { years = 3 }
	cooldown_against_recipient = { years = 5 }
	
	is_valid_showing_failures_only = {
		trigger_if = { # Can't request on titles you already have a claim on
			limit = {
				scope:target ?= {
					any_claimant = { this = scope:puppet_or_actor }
				}
			}
			custom_description = {
				text = buy_claim_has_pressed_claim_already
				always = no
			}
		}
		trigger_if = { # Can't request claims on baronies
			limit = {
				scope:target ?= { tier = tier_barony }
			}
			custom_description = {
				text = buy_claim_has_pressed_claim_already
				always = no
			}
		}
		trigger_if = { # Can't buy Kingdom or Empire Claims if you're a King
			limit = {
				scope:target ?= { tier = tier_kingdom }
				scope:puppet_or_actor.highest_held_title_tier >= tier_kingdom
			}
			custom_description = {
				text = buy_claim_no_king_or_emperor_kingdom
				always = no
			}
		}
		trigger_if = { # Can't buy Empire Claims if you're an Emperor
			limit = {
				scope:target ?= { tier >= tier_empire }
				scope:puppet_or_actor.highest_held_title_tier >= tier_empire
			}
			custom_description = {
				text = buy_claim_no_emperor_empire
				always = no
			}
		}
		trigger_if = { # Can't buy claims on head of faith titles
			limit = {
				scope:target ?= { is_head_of_faith = yes }
			}
			custom_description = {
				text = buy_claim_no_religious_titles
				always = no
			}
		}
		trigger_if = {
			limit = {
				scope:target ?= { has_clerical_region = yes }
			}
			custom_description = {
				text = buy_claim_no_religious_titles
				always = no
			}
		}
		trigger_if = { # Can't buy claims on noble families
			limit = {
				scope:target ?= { is_noble_family_title = yes }
			}
			custom_description = {
				text = buy_claim_no_noble_family_titles
				always = no
			}
		}
		trigger_if = { # Can't buy claims on landless adventurer titles
			limit = {
				scope:target ?= { has_variable = adventurer_creation_reason }
			}
			custom_description = {
				text = buy_claim_no_landless_adventurer_title
				always = no
			}
		}
		scope:puppet_or_actor = { # Variable piety cost depending on tier
			trigger_if = {
				limit = {
					exists = scope:target
					scope:target = {
						tier = tier_county
					}
				}
				piety >= {
					value = minor_buy_claim_piety_value
					if = {
						limit = {
							government_has_mechanic = administrative
						}
						multiply = 1.5
					}
				}
			}
			trigger_if = {
				limit = {
					exists = scope:target
					scope:target = {
						tier = tier_duchy
					}
				}
				piety >= {
					value = medium_buy_claim_piety_value
					if = {
						limit = {
							government_has_mechanic = administrative
						}
						multiply = 1.5
					}
				}
			}
			trigger_if = {
				limit = {
					exists = scope:target
					scope:target = {
						tier = tier_kingdom
					}
				}
				piety >= {
					value = major_buy_claim_piety_value
					if = {
						limit = {
							government_has_mechanic = administrative
						}
						multiply = 1.5
					}
				}
			}
			trigger_if = {
				limit = {
					exists = scope:target
					scope:target = {
						tier = tier_empire
					}
				}
				piety >= {
					value = massive_buy_claim_piety_value
					if = {
						limit = {
							government_has_mechanic = administrative
						}
						multiply = 1.5
					}
				}
			}
			trigger_if = {
				limit = {
					exists = scope:target
					scope:target = {
						tier = tier_hegemony
					}
				}
				piety >= {
					value = monumental_buy_claim_piety_value
					if = {
						limit = {
							government_has_mechanic = administrative
						}
						multiply = 1.5
					}
				}
			}
		}
		scope:recipient = { NOT = { has_strong_hook = scope:puppet_or_actor } }
		tgp_blocked_action_against_tenno_trigger = {
			ACTOR = scope:puppet_or_actor
			TARGET = scope:recipient
		}
	}

	auto_accept = yes

	on_accept = {
		scope:puppet_or_actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = buy_claim_interaction.message
				left_icon = scope:puppet_or_actor
				right_icon = scope:recipient
				add_piety = {
					value = 0
					if = {
						limit = {
							scope:target = {
								tier = tier_county
							}
						}
						add = minor_buy_claim_piety_loss
					}
					if = {
						limit = {
							scope:target = {
								tier = tier_duchy
							}
						}
						add = medium_buy_claim_piety_loss
					}
					if = {
						limit = {
							scope:target = {
								tier = tier_kingdom
							}
						}
						add = major_buy_claim_piety_loss
					}
					if = {
						limit = {
							scope:target = {
								tier = tier_empire
							}
						}
						add = massive_buy_claim_piety_loss
					}
					if = {
						limit = {
							scope:target = {
								tier = tier_hegemony
							}
						}
						add = monumental_buy_claim_piety_loss
					}
					if = {
						limit = {
							government_has_mechanic = administrative
						}
						multiply = 1.5
					}
				}
				add_unpressed_claim = scope:target
			}
			stress_and_fulfillment_impact = {
				just = medium_stress_impact_gain
				honest = medium_stress_impact_gain
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = buy_claim_interaction.message_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:puppet_or_actor
				modifier = claimed_my_title_opinion
				opinion = -60
			}
			trigger_event = perk_interaction.0002
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
