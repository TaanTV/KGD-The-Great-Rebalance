#Interactions relating to vassals

grant_vassal_interaction = {
	category = interaction_category_vassal
	common_interaction = no
	interface = transfer_vassal
	icon = icon_vassal

	special_interaction = grant_vassal_interaction

	populate_actor_list = {
		scope:puppet_or_actor = {
			every_vassal = {
				add_to_list = characters
			}
		}
	}

	desc = grant_vassal_interaction_desc

	prompt = GRANT_VASSAL_SELECT_VASSAL_TO_TRANSFER

	greeting = positive
	notification_text = GRANT_VASSAL_PROPOSAL

	pre_answer_yes_key = ACCEPT
	pre_answer_no_key = DECLINE

	is_shown = {
		scope:recipient != scope:puppet_or_actor
		OR = {
			AND = {
				exists = scope:recipient.liege
				scope:recipient.liege ?= scope:puppet_or_actor
			}
			AND = {
				exists = scope:puppet_or_actor.liege
				scope:puppet_or_actor.liege ?= scope:recipient
			}
			AND = {
				scope:puppet_or_actor.faith.religious_head ?= scope:recipient
				scope:recipient = {
					top_liege = this
				}
			}
		}
		scope:recipient.highest_held_title_tier > tier_barony
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			is_landed = yes # Landless rulers should not have Vassals
			# Gallivanters won't accept additional responsibilities.
			NOT = { has_trait = gallivanter }
		}
		NOT = { scope:puppet_or_actor = { is_at_war_with = scope:recipient } }
		trigger_if = {
			limit = { exists = scope:secondary_actor }
			NOT = { scope:secondary_actor = { is_at_war_with = scope:recipient } }
			scope:recipient.primary_title.tier > scope:secondary_actor.primary_title.tier
		}
		trigger_if = {
			limit = {
				scope:puppet_or_actor = { is_ai = yes }
				exists = scope:secondary_actor
			}
			trigger_if = {
				limit = {
					NOT = {
						scope:recipient.primary_title = {
							any_in_de_jure_hierarchy = {
								holder = scope:secondary_actor
							}
						}
					}
				}
				scope:recipient.vassal_count < scope:recipient.vassal_limit # Avoid creating vassals with unsustainable situations
			}
			scope:secondary_actor = {
				NOT = { has_character_flag = ai_should_not_transfer }
			}
		}
		trigger_if = {
			limit = { scope:recipient.highest_held_title_tier <= tier_county }
			scope:recipient = { highest_held_title_tier > tier_county }
		}
		trigger_else = {
			scope:puppet_or_actor = {
				custom_description = {
					text = "basic_valid_vassal_to_tranfer"
					object = scope:recipient
					any_vassal = {
						NOT = { highest_held_title_tier = tier_barony }
						scope:recipient.highest_held_title_tier > highest_held_title_tier
					}
				}
			}
		}
		# Prevent Admin AI transferring admin vassals under non-admin vassals
		NAND = {
			scope:puppet_or_actor ?= {
				is_ai = yes
				government_has_mechanic = administrative
			}
			scope:recipient ?= {
				NOT = { government_has_mechanic = administrative }
			}
			scope:secondary_actor ?= { government_has_mechanic = administrative }
		}
		# Ritsuryo cannot transfer Soryo to Ritsuryo
		custom_tooltip = {
			text = ritsuryo_to_soryo_invalid_to_transfer
			NAND = {
				scope:puppet_or_actor ?= { has_government = japan_administrative_government }
				scope:recipient ?= { has_government = japan_administrative_government }
				scope:secondary_actor ?= { has_government = japan_feudal_government }
			}
		}
		custom_tooltip = {
			text = cannot_transfer_bloc_leader_as_vassal_tt
			NOT = {
				scope:secondary_actor ?= {
					exists = house.house_confederation.leading_house
					house.house_confederation.leading_house ?= house
					is_house_head = yes
				}
			}
		}
		
		#ADDED - cant transfer vasssals that are in angy factions (NOTE: may need a custom_tooltip)
		trigger_if = {
			limit = {
				exists = scope:secondary_actor
				scope:secondary_actor = { is_a_faction_member = yes }
			}
			scope:secondary_actor.joined_faction = { discontent_per_month <= 0 }
		}
		
		#scope:secondary_actor ?= {
		#	trigger_if = {
		#		limit = { is_a_faction_member = yes }
		#	}
		#}
	}

	can_be_picked = {
		scope:recipient != root
		highest_held_title_tier > tier_barony
		highest_held_title_tier < scope:recipient.highest_held_title_tier

		trigger_if = {
			limit = { scope:puppet_or_actor.liege = scope:recipient }
			custom_tooltip = {
				text = is_de_jure_liege_of_target
				NOT = {
					primary_title = {
						any_this_title_or_de_jure_above = {
							holder = scope:puppet_or_actor
						}
					}
				}
			}
		}

		# You can't give your diarch away as a vassal.
		custom_description = {
			text = cannot_grant_own_diarch_as_vassal
			subject = scope:puppet_or_actor.diarch
			NOT = { scope:puppet_or_actor.diarch ?= root }
		}

		# Not at war with the vassal being transferred (blocks explicit transfer of war enemies, which invalidates the war).
		custom_description = {
			text = "basic_is_at_war_with_vassal"
			subject = root
			object = scope:puppet_or_actor
			NOT = {
				root = {
					is_at_war_with = scope:puppet_or_actor
				}
			}
		}

		# Not at war with a vassal under the vassal being transferred (blocks implicit transfers of war enemies).
		custom_description = {
			text = "basic_is_at_war_with_sub_vassal"
			subject = root
			object = scope:puppet_or_actor
			NOT = {
				scope:puppet_or_actor = {
					any_vassal_or_below = {
						is_at_war_with = root
					}
				}
			}
		}

		# EP3 Admin - You cannot give away noble families
		custom_description = {
			text = admin_cannot_grant_vassal_noble_family
			subject = root
			any_held_title = {
				is_noble_family_title = no
			}
		}
	}

	auto_accept = {
		scope:recipient = {
			custom_description = {
				text = wants_more_vassals
				is_ai = yes
			}
		}
	}

	on_accept = {
		#Holy Order patronage loss warning
		if = {
			limit = {
				#Holy order is actually involved
				scope:puppet_or_actor = {
					any_patroned_holy_order = {
						leader.capital_county.holder ?= {
    						target_is_same_character_or_above = scope:secondary_actor
						}
    				}
				}
				scope:recipient = {
					highest_held_title_tier >= tier_duchy
				}
			}
			scope:puppet_or_actor = {
				custom_tooltip = holy_order_control_loss_warning_tt
			}
		}

		scope:puppet_or_actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = grant_vassal_interaction_notification
				right_icon = scope:recipient
				left_icon = scope:secondary_actor

				# If we're granting a vassal to our liege, gain a truce with our former vassal (we can't immediately reconquer them, and they can't attack us either).
				if = {
					limit = {
						scope:puppet_or_actor = {
							is_vassal_of = scope:recipient
						}
					}
					add_truce_both_ways = {
						character = scope:secondary_actor
						days = 3650
						name = TRUCE_GRANT_INDEPENDENCE
					}
				}

				# Transfer the vassal over to their new liege.
				create_title_and_vassal_change = {
					type = granted
					save_scope_as = change
					add_claim_on_loss = no
				}
				scope:secondary_actor = {
					change_liege = {
						liege = scope:recipient
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change

				# Gain an opinion bonus with the person we gave a vassal to.
				scope:recipient = {
					add_opinion = {
						target = scope:puppet_or_actor
						modifier = granted_vassal
					}
				}
			}
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = miniscule_unity_gain
			DESC = clan_unity_grant_vassal.desc
			REVERSE_NON_HOUSE_TARGET = yes
		}

		hidden_effect = {
			# Struggle catalyst
			if = {
				limit = {
					fp3_struggle_involves_one_supporter_and_one_detractor = {
						FIRST = scope:puppet_or_actor
						SECOND = scope:recipient
					}
					scope:puppet_or_actor = {
						any_character_struggle	 = {
							involvement = involved
							phase_has_catalyst = catalyst_grants_vassal_to_de_jure_liege_supporter_detractor
							is_secondary_character_involvement_involved_trigger = {
								CHAR = scope:recipient
							}
						}
					}
					scope:recipient.primary_title = {
						any_this_title_or_de_jure_above = {
							holder = scope:secondary_actor
						}
					}
				}
				scope:puppet_or_actor = {
					every_character_struggle = {
						involvement = involved
						activate_struggle_catalyst = {
							catalyst = catalyst_grants_vassal_to_de_jure_liege_supporter_detractor
							character = scope:puppet_or_actor
						}
						log_debug_variable_for_persian_struggle_effect = { VAR = concession_catalyst_grants_vassal_to_de_jure_liege_supporter_detractor }
					}
				}
			}
			if = {
				limit = {
					scope:recipient = {
						government_has_flag = government_is_nomadic
					}
				}
				while = {
					limit = {
						scope:secondary_actor ?= {
							any_held_title = {
								vassal_grant_de_jure_shift_trigger = yes
							}
						}
					}
					scope:secondary_actor ?= {
						random_held_title = {
							limit = {
								vassal_grant_de_jure_shift_trigger = yes
							}
							set_de_jure_liege_title = scope:recipient.primary_title
						}
					}
				}
			}
		}
		#Opinion loss with underlings who think they should get the vassal you're granting
		if = {
			limit = {
				#AI will usually grant title to de jure holders, it's players who could use this steering
				scope:puppet_or_actor = {
					is_ai = no
				}
			}
			scope:secondary_actor = {
				if = {
					limit = {
						any_held_title = {
							any_claimant = {
								this = scope:recipient
							}
						}
					}
					ordered_held_title = {
						order_by = tier
						limit = {
							any_claimant = {
								this = scope:recipient
							}
						}
						save_scope_as = highest_claim
					}
					scope:recipient = { claimant_jealousy_immunity_effect = yes }
				}
				every_held_title = {
					limit = {
						tier >= tier_county
						tier = scope:secondary_actor.primary_title.tier
					}
					save_scope_as = title_transfered
					scope:puppet_or_actor = {
						title_granting_controversy_effect = {
							TITLE = scope:title_transfered
							INTERACTION = grant_vassal_interaction
						}
					}
				}
				every_in_list = {
					list = strong_claimants
					limit = {
						NOT = { this = scope:recipient }
					}
					save_scope_as = claimer
					scope:secondary_actor = {
						ordered_held_title = {
							limit = {
								any_claimant = {
									this = scope:claimer
								}
							}
							save_scope_as = highest_denied_claim
						}
					}
					claimant_rivalry_mercy_effect = yes
				}
			}
			#Being given to a non de jure liege should piss vassals off
			if = {
				limit = {
					NOT = {
						scope:secondary_actor = {
							any_held_title = {
								any_this_title_or_de_jure_above = {
									holder ?= scope:recipient
								}
							}
						}
					}
				}
				scope:secondary_actor = {
					add_opinion = {
						modifier = subordinated_to_non_de_jure_liege_opinion
						target = scope:puppet_or_actor
						opinion = -30 
					}
				}
			}
			
		}
	}

	on_decline = {
		scope:puppet_or_actor = {
			trigger_event = char_interaction.0211
		}

		# If we're a clan this interaction affects unity
		add_clan_unity_interaction_effect = {
			CHARACTER = scope:recipient
			TARGET = scope:puppet_or_actor
			VALUE = miniscule_unity_loss
			DESC = clan_unity_grant_vassal_decline.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
	}

	# AI handled entirely though code
}

retract_vassal_interaction = {
	category = interaction_category_vassal
	special_interaction = retract_vassal_interaction
	interface = transfer_vassal
	ai_maybe = yes
	ai_min_reply_days = 4
	ai_max_reply_days = 9
	popup_on_receive = yes
	pause_on_receive = yes
	icon = icon_liege

	interface_priority = 109 #after revoke title

	desc = retract_vassal_interaction_desc

	on_decline_summary = general_rebellion_decline_summary

	prompt = RETRACT_VASSAL_SELECT_VASSAL_TO_RETRACT

	notification_text = RETRACT_VASSAL_PROPOSAL

	populate_recipient_list = {
		scope:recipient = {
			every_vassal = {
				add_to_list = characters
			}
		}
	}

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			has_realm_law_flag = title_revocation_allowed
			primary_title.tier > tier_county
		}
	}

	is_shown = {
		scope:recipient != scope:puppet_or_actor
		scope:recipient = {
			highest_held_title_tier > tier_barony
			liege ?= scope:puppet_or_actor
		}
	}

	is_valid_showing_failures_only = {
		vassal_retraction_standard_valid_showing_failures_only_trigger = yes
		# Make sure this exactly matches "can_be_picked" section below
		scope:recipient = {
			custom_description = {
				text = "basic_vassal_to_retract"
				object = scope:actor
				any_vassal = {
					highest_held_title_tier > tier_barony
				}
			}
		}
		# Plus make sure there's a legal right to retract.
		scope:puppet_or_actor = {
			trigger_if = {
				limit = {
					government_has_flag = government_is_tribal
				}
				custom_description = {
					text = "has_law_allowing_vassal_retraction_tribal"
					has_realm_law_flag = vassal_retraction_allowed
				}
			}
			trigger_else_if = {
				limit = {
					government_has_flag = government_is_administrative
				}
				custom_description = {
					text = "has_law_allowing_vassal_retraction_admin"
					has_realm_law_flag = vassal_retraction_allowed
				}
			}
			trigger_else_if = {
				limit = {
					realm_law_use_church_authority = yes
				}
				custom_description = {
					text = "has_law_allowing_vassal_retraction_church"
					has_realm_law_flag = vassal_retraction_allowed
				}
			}
			trigger_else = {
				custom_description = {
					text = "has_law_allowing_vassal_retraction_crown"
					has_realm_law_flag = vassal_retraction_allowed
				}
			}
			#ADDED
			underage_diarch_loyalty_restriction = yes
		}
		scope:recipient = {
			NOT = { has_strong_hook = scope:puppet_or_actor }
		}
		custom_tooltip = {
			text = cannot_take_overt_hostile_actions_against_diarch.tt
			NOT = { scope:recipient ?= scope:puppet_or_actor.diarch }
		}
	}

	can_send = {
		scope:puppet_or_actor = {
			custom_description = {
				text = "character_interactions_hostile_actions_disabled_delay"
				NOT = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
		}
		
		#ADDED - cant provoke revolts while your troops are in their lands, unless its autoaccept.
		trigger_if = {
			limit = { 
				scope:puppet_or_actor = { is_ai = no }
				scope:recipient = {
					is_imprisoned = no
					NAND = {
						scope:hook = yes
						scope:puppet_or_actor = { has_strong_hook = scope:recipient }
					}
				}
			}
			puppet_or_actor_doesnt_have_troops_in_recipient_lands = yes
		}
	}

	can_be_picked = {
		highest_held_title_tier > tier_barony
	}

	cost = {
		# Usually, this action is free, but during certain diarchies, we want it to cost.
		prestige = {
			value = liege_diarchy_surcharge_interaction_retract_vassal_interaction_value

			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	on_send = {
		scope:puppet_or_actor = {
			add_character_flag = {
				flag = flag_hostile_actions_disabled_delay
				days = 10
			}
		}
		hidden_effect = {
			scope:secondary_recipient = {
				add_character_flag = {
					flag = ai_should_not_transfer
					years = 3
				}
			}
		}
	}

	on_accept = {
		retract_vassal_interaction_effect = yes

		if = {
			limit = {
				scope:puppet_or_actor = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
			scope:puppet_or_actor = { remove_character_flag = flag_hostile_actions_disabled_delay }
		}

		scope:puppet_or_actor = {
			trigger_event = char_interaction.0220
		}
	}

	on_decline = {
		scope:puppet_or_actor = { vassal_revocation_tyranny_effect = yes }
		
		#ADDED - this if and put the war into ELSE
		if = {
			limit = {
				scope:puppet_or_actor = { has_legitimacy_flag = vassals_refuse_retract_with_no_war }
			}
			scope:puppet_or_actor = { 
				add_prestige = medium_prestige_loss
			}
		}
		else = {
			war_for_retraction_effect = {
				LIEGE = scope:puppet_or_actor
				REBEL = scope:recipient
				SUBVASSAL = scope:secondary_recipient
			}
		}
		scope:recipient = {
			add_opinion = {
				target = scope:puppet_or_actor
				modifier = retracted_vassal
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = { has_realm_law_flag = vassal_refusal_is_treason }
			}
			scope:puppet_or_actor = {
				add_opinion = {
					target = scope:recipient
					modifier = treasonous_retract_refusal
				}
			}
		}
		if = {
			limit = {
				scope:puppet_or_actor = { has_character_flag = flag_hostile_actions_disabled_delay }
			}
			scope:puppet_or_actor = { remove_character_flag = flag_hostile_actions_disabled_delay }
		}
	}

	auto_accept = {
		OR = {
			scope:recipient = {
				#is_imprisoned_by = scope:puppet_or_actor
				#ADDED - just check the general imprisonment
				is_imprisoned = yes
			}
			custom_description = {
				text = "spending_hook"
				subject = scope:puppet_or_actor
				object = scope:recipient
				scope:hook = yes
				scope:puppet_or_actor = { has_strong_hook = scope:recipient }
			}
		}
	}

	can_be_blocked = {
		custom_description = { #Hook from Recipient can counteract, but only if Actor is not forcing the action via Strong Hook.
			text = "spending_hook"
			subject = scope:recipient
			object = scope:puppet_or_actor
			scope:recipient = { has_hook = scope:puppet_or_actor }
			NAND = {
				scope:hook = yes
				scope:puppet_or_actor = { has_strong_hook = scope:recipient }
			}
		}
	}

	on_blocked_effect = {
		scope:recipient = {
			remove_hook = {
				target = scope:puppet_or_actor
			}
		}
	}

	ai_accept = {
		base = 0 # Try to make it 0 for most interactions

		legalism_lawful_authority_acceptance_modifier = {
			ACTOR = scope:puppet_or_actor
		}

		modifier = {
			has_game_rule = hard_difficulty
			scope:actor = { is_ai = no }
			add = -50
			desc = FACTION_REASON_DIFFICULTY_HARD
		}

		modifier = {
			has_game_rule = very_hard_difficulty
			scope:actor = { is_ai = no }
			add = -100
			desc = FACTION_REASON_DIFFICULTY_VERY_HARD
		}

		modifier = {
			has_game_rule = hard_difficulty
			scope:actor = { is_ai = yes }
			add = 50
		}

		modifier = {
			has_game_rule = very_hard_difficulty
			scope:actor = { is_ai = yes }
			add = 100
		}

		modifier = {
			scope:hook = yes
			add = 40
			desc = SCHEME_WEAK_HOOK_USED
		}

		opinion_modifier = { # Opinion Factor
			who = scope:recipient
			opinion_target = scope:puppet_or_actor
			multiplier = 1.0
			desc = AI_OPINION_REASON
		}

		modifier = {
			add = 25
			scope:puppet_or_actor = {
				has_realm_law_flag = vassal_refusal_is_treason
			}
			desc = AI_REFUSAL_IS_TREASON
		}

		modifier = {
			add = 20
			scope:recipient = {
				has_trait = trusting
			}
			desc = TAKE_THE_VOWS_TRUSTING
		}

		modifier = {
			add = 20
			scope:recipient = {
				has_trait = content
			}
			desc = INTERACTION_CONTENT
		}

		modifier = {
			add = 20
			scope:puppet_or_actor.current_military_strength > scope:recipient.current_military_strength
			scope:recipient = {
				has_trait = craven
			}

	}

	on_decline_summary = stop_attacker_vassal_war_decline_summary

	is_shown = {
		scope:recipient != scope:actor
		scope:recipient = {
			liege ?= scope:actor
			highest_held_title_tier > tier_barony
			any_character_war = {
				primary_attacker ?= scope:recipient
				save_temporary_scope_as = target_war
			}
		}
		scope:actor = {
			NOT = { is_participant_in_war = scope:target_war }
		}
	}

	is_valid_showing_failures_only = {
		custom_description = {
			text = cannot_stop_war_when_already_fighting
			object = scope:recipient
			scope:recipient = {
				NOT = {
					any_war_enemy = {
						this = scope:actor
					}
				}
			}
		}
		custom_description = {
			text = cannot_stop_war_timer
			object = scope:recipient
			#ADDED - OR and the bottom part of it
			OR = {
				NOT = {
					scope:recipient = {
						has_opinion_modifier = {
							target = scope:actor
							modifier = attempt_to_force_end_war_opinion
						}
					}
				}
				scope:recipient = { is_imprisoned_by = scope:actor }
			}
		}
		NOT = {
			scope:recipient = {
				has_strong_hook = scope:actor
			}
		}
	}

	can_be_picked = {
		scope:target.primary_attacker = scope:recipient
		#ADDED
		scope:target = {
			NOR = {
				using_cb = undirected_great_holy_war
				using_cb = directed_great_holy_war
			}
		}
	}

	on_accept = {
		scope:recipient = {
			add_opinion = {
				modifier = attempt_to_force_end_war_opinion
				target = scope:actor
			}
		}
		hidden_effect = {
			#ADDED - now an IF, inside of IF is ADDED too.
			if = {
				limit = {
					scope:recipient = {
						is_imprisoned = yes
						is_imprisoned_by = scope:actor
					}
				}
				scope:target = {
					primary_defender = {
						save_scope_as = defender
					}
				}
				scope:actor = {
					trigger_event = vassal_interaction.0002 #auto-accept
				}
			}
			else = {
				scope:recipient = {
					trigger_event = vassal_interaction.0001 #Starts "negotiations"
				}
			}
		}
		show_as_tooltip = {
			scope:target ?= { end_war = white_peace }
		}

		# If we're a clan this interaction affects unity - But we only show it in the tooltip, actual effect is applied in the respective follow-up events
		show_as_tooltip = {
			if = {
				limit = { # If we are asking a non-house member to stop attacking a house member, gain unity
					exists = scope:recipient.house
					exists = scope:actor.house
					scope:recipient.house != scope:actor.house
				}
				add_clan_unity_interaction_effect = {
					CHARACTER = scope:actor
					TARGET = scope:target.primary_defender
					VALUE = medium_unity_gain
					DESC = clan_unity_aided_in_war.desc
					REVERSE_NON_HOUSE_TARGET = no
				}
			}
			else_if = {
				limit = { # If we are asking a house member to stop attacking a non-house member, lose unity
					exists = scope:actor.house
					exists = scope:target.primary_defender.house
					scope:target.primary_defender.house != scope:actor.house
				}
				add_clan_unity_interaction_effect = {
					CHARACTER = scope:actor
					TARGET = scope:recipient
					VALUE = medium_unity_loss
					DESC = clan_unity_stopped_vassal_war.desc
					REVERSE_NON_HOUSE_TARGET = no
				}
			}
		}
	}

	on_decline = {
		show_as_tooltip = {
			scope:recipient = {
				add_opinion = {
					modifier = attempt_to_force_end_war_opinion
					target = scope:actor
				}
			}
			scope:actor = { add_prestige = minor_prestige_loss }
		}
	}

	auto_accept = yes
}
