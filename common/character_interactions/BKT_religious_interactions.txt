request_excommunication_interaction = {
	category = interaction_category_hostile
	icon = excommunication
	interface_priority = 30

	desc = request_excommunication_interaction_desc
	redirect = {
		#excommunication victim needs to be made secondary recipient
		scope:recipient = {
			save_scope_as = secondary_recipient
		}
		scope:puppet_or_actor = {
			petition_head_of_faith_save_religious_authority_effect = { SCOPE = excommunication_petition_authority }
		}

		# Are we looking for a territorial cleric or going to the target's shared Head of Faith?
		if = {
			limit = {
				OR = {
					scope:secondary_recipient = {
						need_hof_for_clergy_interaction_trigger = no
					}
					trigger_if = {
						limit = { exists = scope:excommunication_petition_authority }
						NOT = {
							scope:secondary_recipient = {
								petition_head_of_faith_religious_authority_is_trigger = {
									CHARACTER = scope:excommunication_petition_authority
								}
							}
						}
					}
					trigger_else = { always = yes }
				}
			}
			#Is it a ruler or just some regular joe?
			if = {
				limit = {
					scope:secondary_recipient = {
						is_ruler = yes
					}
				}
				pam_redirect_request_excommunication_ruler_effect = yes
			}
			else_if = {
				limit = {
					exists = scope:secondary_recipient.liege
				}
				pam_redirect_request_excommunication_courtier_effect = yes
			}
			else = {
				pam_redirect_request_excommunication_wanderer_effect = yes
			}
		}
		else = {
			scope:excommunication_petition_authority ?= {
				save_scope_as = recipient
			}
		}
		
		#Still hasn't been set
		#stop errors from the above process failing, this should be blocked anyway
		if = {
			limit = {
				scope:secondary_recipient = scope:recipient
			}
			clear_saved_scope = recipient
		}
	}

	cooldown_against_recipient = { years = 3 }

	notification_text = RELIGIOUS_HEAD_REQUEST_EXCOMMUNICATION

	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			piety >= major_piety_value
			is_adult = yes
			NOT = { has_character_flag = hof_request_cooldown }
			faith ?= { faith_has_sacraments_central_trigger = yes }
		}
	}

	is_shown = {
		scope:puppet_or_actor.faith ?= {
			faith_has_sacraments_central_trigger = yes
			OR = {
				petition_head_of_faith_faith_qualifies_trigger = yes
				has_doctrine_parameter = has_clerical_regions
				AND = {
					has_doctrine = doctrine_spiritual_head
					exists = religious_head
				}
				any_faith_ruler = {
					government_has_flag = government_is_theocracy
					highest_held_title_tier >= tier_duchy
				}
			}
			this = scope:secondary_recipient.faith
		}
		NOR = {
			scope:puppet_or_actor ?= scope:secondary_recipient
			scope:puppet_or_actor.faith.religious_head ?= scope:puppet_or_actor
			scope:puppet_or_actor.faith.religious_head ?= scope:secondary_recipient
			AND = {
				NOT = { exists = scope:puppet_or_actor.faith.religious_head }
				OR = {
					scope:secondary_recipient.rite.head_of_rite ?= scope:puppet_or_actor
					scope:secondary_recipient.rite.head_of_rite ?= scope:secondary_recipient
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		# The local path still needs a valid territorial cleric.
		trigger_if = {
			limit = {
				NOT = {
					scope:secondary_recipient = {
						petition_head_of_faith_religious_authority_is_trigger = {
							CHARACTER = scope:recipient
						}
					}
				}
			}
			custom_tooltip = {
				text = no_valid_excommunicator_tt
				scope:puppet_or_actor.faith = {
					any_faith_subordinate = {
						highest_held_title_tier >= tier_duchy
						save_temporary_scope_as = cleric_temp
						scope:secondary_recipient = {
							pam_valid_excommunication_target_trigger = { EXCOMMUNICATOR = scope:cleric_temp }
						}
					}
				}
			}
			scope:secondary_recipient.highest_held_title_tier < tier_empire
		}
		# additional checks if not requested through puppet...
		trigger_if = {
			limit = { NOT = { scope:puppet_or_actor ?= scope:actor } }
			scope:secondary_recipient = {
				NOT = { has_strong_hook = scope:puppet_or_actor }
			}
		}
		trigger_if = {
			limit = {
				petition_head_of_faith_excommunication_target_requires_petition_trigger = yes
				NOT = { scope:excommunication_hook = yes }
			}
			scope:puppet_or_actor = {
				piety >= massive_piety_value
			}
		}
		trigger_else = {
			scope:puppet_or_actor = {
				piety >= request_excommunication_piety_cost_value
			}
		}

		# Excommunication target
		scope:secondary_recipient = {
			NOR = {
				has_trait = excommunicated
				has_character_modifier = excommunication_recently_lifted
			}

	on_accept = {
		#If you are trying to excommunicate a duke or below, then it's a simple process of talking to the correct clergy above them
		if = {
			limit = {
				OR = {
					AND = {
						scope:secondary_recipient.highest_held_title_tier < tier_empire
						has_pam_dlc_trigger = no
					}
					#otherwise kings and above you must travel to rome to request excommunication
					AND = {
						scope:secondary_recipient.highest_held_title_tier < tier_kingdom
						has_pam_dlc_trigger = yes
					}
				}
			}
			if = {
				limit = {
					scope:secondary_recipient = { is_alive = yes }
				}
				if = {
					limit = { scope:excommunication_hook = yes }
					scope:puppet_or_actor = {
						use_hook = scope:recipient
					}
				}

				scope:puppet_or_actor = {
					#Spend piety for the excommunication.
					#Peace of God makes it free against aggressors though
					if = {
						limit = {
							NAND = {
								rite = {
									rite_has_parameter = tenet_peace_of_god_excommunication
								}
								has_opinion_modifier = {
									modifier = declared_war
									target = scope:secondary_recipient
								}
							}
						}
						add_piety = {
							value = {
								value = request_excommunication_piety_cost_value
								multiply = -1
							}
						}
					}				

					#Letter event to inform character the pope agreed to their request.
					trigger_event = {
						id = religious_interaction.1022
					}
					if = {
						limit = {
							any_character_situation = {
								situation_type = the_christian_church
							}
							situation:the_christian_church ?= {
								situation_top_has_catalyst = catalyst_request_excommunication
							}
						}
						if = {
							limit = {
								scope:secondary_recipient = {
									highest_held_title_tier >= tier_kingdom
								}
							}
							situation:the_christian_church = { trigger_situation_catalyst = catalyst_request_excommunication }
						}
						else = {
							situation:the_christian_church = { trigger_situation_catalyst = catalyst_request_excommunication_minor }
						}
					}
				}
				excommunicate_character = {
					EXCOMMUNICATOR = scope:recipient
					TARGET_CHARACTER = scope:secondary_recipient
					REQUESTING_CHARACTER = scope:puppet_or_actor
				}
				scope:secondary_recipient = {
					# Excommunication letter (same 5-day delay as direct papal excommunication).
					trigger_event = {
						id = religious_interaction.1020
						days = 5
					}
				}
				# If we're a clan this interaction affects unity
				add_clan_unity_interaction_effect = {
					CHARACTER = scope:puppet_or_actor
					TARGET = scope:recipient
					VALUE = medium_unity_loss
					DESC = clan_unity_request_excommunication.desc
					REVERSE_NON_HOUSE_TARGET = no
				}
				hof_request_accept_consequences_effect = {
					HOF = scope:recipient
				}
			}
		}
		# King+ targets require a petition trip to Rome (same pattern as lifting a papal excommunication).
		else = {
			scope:puppet_or_actor = {
				save_scope_as = excom_request
			}
			custom_tooltip = petition_head_of_faith_excommunication_setup_interaction_accept_tt
			petition_head_of_faith_prepare_petition_effect = {
				TYPE = request_excommunication
				HOOK = excommunication_hook
				PIETY_VALUE = massive_piety_value
			}
		}	
	}
