	on_accept = {
		scope:recipient = {
			set_variable = {
				name = invite_to_court_cooldown
				years = 5
			}
			# Message relevant parties
			liege ?= {
				if = {
					limit = {
						NOT = { government_has_flag = government_is_merchant }
					}
					send_interface_message = {
						type = event_generic_neutral
						title = msg_courtier_left_court_interaction_title
						right_icon = scope:recipient
						show_as_tooltip = { remove_courtier_or_guest = scope:recipient }
					}
				}
				else = {
					send_interface_message = {
						type = event_generic_bad
						title = msg_courtier_left_court_interaction_merchant_title
						right_icon = scope:recipient
						show_as_tooltip = { remove_courtier_or_guest = scope:recipient }
					}
				}
				# House Relation impact
				if = { # Only change house relations w vassals or below
					limit = { this != scope:actor.liege }
					house ?= {
						if = {
							limit = { this != scope:actor.house }
							if = {
								limit = {
									NOT = { has_house_relation_with = scope:actor.house }
								}
								set_house_relation = {
									target = scope:actor.house
									level = quarrel
									description = house_relation_poached_talent_desc
								}
							}
							else = {
								change_house_relation_effect = {
									HOUSE = scope:actor.house
									VALUE = house_relation_damage_minor_value
									REASON = poached_talent
									CHAR = scope:actor
									TARGET_CHAR = scope:recipient.liege
									TITLE = scope:dummy_gender
								}
							}
						}
					}
				}
			}
		}

		scope:actor = {
			send_interface_message = {
				type = event_generic_neutral
				title = invite_to_court_interaction_notification
				right_icon = scope:recipient

				scope:recipient = {
					every_traveling_family_member = {
						scope:actor = { add_courtier = prev }
						hidden_effect = { return_to_court = yes }
					}
				}
				if = {
					limit = { always = scope:hook }
					scope:actor = { use_hook = scope:recipient }
				}
				if = {
					limit = { government_has_flag = government_uses_company_sentiment }

						add_unity_value = {
							value = minor_unity_loss
							character = scope:recipient
							desc = merchant_invite_associate_loss
						}
					}
				}
			}
		}

					}
					target = scope:recipient
				}
				stress_and_fulfillment_impact = {
					greedy = minor_stress_impact_gain
				}
			}
		}
