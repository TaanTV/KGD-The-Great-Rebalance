hire_laamp_mercenaries_interaction = {
	category = interaction_category_diplomacy
	icon = alliance
	desc = hire_laamp_mercenaries_interaction_desc

	greeting = positive
	notification_text = hire_laamp_mercenaries_interaction_notification
	popup_on_receive = yes
	pause_on_receive = yes

	interface_priority = 60
	ai_min_reply_days = 0
	ai_max_reply_days = 0

	cooldown_against_recipient = { years = 3 }
	can_send_despite_rejection = yes
	ai_accept_negotiation = yes
	send_options_exclusive = no
	is_available = {
		trigger_if = {
			limit = { is_ai = yes }
			debt_level <= 1
			any_character_war = {
				is_war_leader = prev
			}
			war_chest_gold >= quarter_ai_war_chest_gold_maximum
			any_war_ally = {
				count < 3
				has_government = landless_adventurer_government
			}
			any_character_war = {
				count = all
				OR = {
					AND = {
						primary_attacker = root
						defender_war_score <= 25	#prev 95
					}
					AND = {
						primary_defender = root
						attacker_war_score <= 25		#prev 95
					}
				}
			}
		}
	}

	is_shown = {
		has_playable_adventurer_dlc_trigger = yes
		scope:recipient = {
			has_government = landless_adventurer_government
		}
		scope:puppet_or_actor ?= {
			this != scope:recipient
			trigger_if = { # Avoid calling Adventurers without armies
				limit = {
					scope:actor = { is_ai = yes }
				}
				scope:recipient.current_military_strength >= 100
				scope:recipient = {
					NOT = {
						has_character_flag = ai_hire_blocker
					}
				}
			}
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = {
			is_travelling = no
			is_at_war = no
			is_imprisoned = no
		}
		scope:puppet_or_actor = {
			custom_tooltip = {
				text = MERCENARY_COMPANY_COST_BANKRUPTCY
				debt_level <= 1
			}
			is_at_war = yes
			custom_tooltip = {
				text = I_MUST_BE_THE_WAR_LEADER
				any_character_war = {
					is_war_leader = scope:puppet_or_actor
				}
			}
			NOT = {
				is_at_war_with = scope:recipient
			}
		}
		scope:puppet_or_actor = {
			NOT = {
				any_war_ally = {
					count >= 3
					has_government = landless_adventurer_government
				}
				custom_tooltip = {
					at_mercenary_limit = yes
					text = MERCENARY_HIRE_LIMIT_REACHED
				}
			}
		}
		scope:puppet_or_actor = {
			OR = {
				scope:actor = { is_ai = no }
				any_character_war = {
					count = all
					OR = {
						AND = {
							primary_attacker = scope:puppet_or_actor
							defender_war_score <= 95
						}
						AND = {
							primary_defender = scope:puppet_or_actor
							attacker_war_score <= 95
						}
					}
				}
			}
		}
		# Adventurers can't join wars in places they're exiled from.
		trigger_if = {
			limit = {
				scope:puppet_or_actor = { top_liege != this }
			}
			custom_tooltip = {
				text = the_wake_1
				NOT = {
					is_laamp_exiled_from_province_trigger = {
						PROVINCE = scope:puppet_or_actor.capital_province
						LAAMP = scope:recipient
					}
				}
			}
		}
	}

	auto_accept = {
		custom_description = {
			text = "spending_hook"
			subject = scope:puppet_or_actor
			object = scope:recipient
			scope:hook = yes
			scope:puppet_or_actor = { has_strong_hook = scope:recipient }
		}
	}

	on_accept = {
		scope:puppet_or_actor = {
			every_character_war = {
				add_to_list = actor_wars
			}
		}
		every_in_list = {
			list = actor_wars
			limit = {
				primary_defender = scope:puppet_or_actor
			}
			add_to_list = join_as_defender
			remove_from_list = actor_wars
		}
		every_in_list = {
			list = actor_wars
			limit = {
				primary_attacker = scope:puppet_or_actor
			}
			add_to_list = join_as_attacker
			remove_from_list = actor_wars
		}

		every_in_list = {
			list = join_as_attacker
			add_attacker = scope:recipient
		}
		every_in_list = {
			list = join_as_defender
			add_defender = scope:recipient
		}
		#list all wars that scope:recipient can't join
		every_in_list = {
			list = actor_wars
			custom = hire_laamp_mercenaries_interaction.not_primary_side
			# I swear it was the only way to make the tooltip look nice
			every_in_list = {
				list = actor_wars
				save_temporary_scope_as = war
				custom_tooltip = hire_laamp_mercenaries_interaction.war
			}
		}
		scope:puppet_or_actor = {
			if = {
				limit = {
					has_treasury = no
				}
				pay_short_term_gold = {
					target = scope:recipient
					gold = ep3_hire_laamp_mercs_cost_value
				}
			}
			else = {
				pay_short_term_treasury = {
					target = scope:recipient
					treasury = ep3_hire_laamp_mercs_cost_value
				}
			}
			add_to_variable_list = {
				name = hired_mercenaries
				target = scope:recipient
			}
			custom_tooltip = hire_laamp_mercenaries_interaction.contribution_pay
			if = {
				limit = {
					has_title = title:h_china
					situation:dynastic_cycle ?= {
						situation_top_has_catalyst = catalyst_hegemon_hired_mercenary
					}
				}
				situation:dynastic_cycle ?= {
					trigger_situation_catalyst = {
						catalyst = catalyst_hegemon_hired_mercenary
						character = scope:puppet_or_actor
					}
				}
			}
		}
		scope:recipient = {
			set_variable = {
				name = joined_as_mercenary
				value = scope:puppet_or_actor
			}
		}
		add_house_unity_interaction_effect = {
			CHARACTER = scope:puppet_or_actor
			TARGET = scope:recipient
			VALUE = medium_unity_gain
			DESC = clan_unity_alliance.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
		#debug_log = "Landed ruler hired a LAAMP to join wars"
		#debug_log_scopes = yes
		#for debug purposes:
		#if = {
		#	limit = { NOT = { exists = global_var:hire_laamp_as_mercs_accepted } }
		#	set_global_variable = {
		#		name = hire_laamp_as_mercs_accepted
		#		value = 0
		#	}

		#else = {
		#	change_global_variable = {
		#		name = hire_laamp_as_mercs_accepted
		#		add = 1
		#	}
		#}
	}

	on_decline = {
		scope:recipient = {
			trigger_event = ep3_interactions_events.0111

			TARGET = scope:recipient
			VALUE = minor_unity_loss
			DESC = clan_unity_alliance_refusal.desc
			REVERSE_NON_HOUSE_TARGET = no
		}
