support_clerical_candidacy_interaction = {
	icon = support_candidacy
	common_interaction = yes
	hidden = yes

	desc = support_clerical_candidacy_interaction_desc

	target_type = title
	target_source = actor
	target_filter = domain_clerical_regions

	can_be_picked_title = {
		# Only the capital see, so the search stays one title wide
		scope:actor.capital_county.clerical_region_title ?= scope:target
		trigger_if = {
			limit = { exists = scope:target.current_heir }
			scope:target.current_heir = {
				NOR = {
					this = scope:actor
					puppeteer ?= scope:actor
					has_relation_friend = scope:actor
					has_relation_lover = scope:actor
					#ADDED - a bit more generous trigger here just so AI doesnt waste as much piety
					is_close_or_extended_family_of = scope:actor
				}
			}
		}
		trigger_if = {
			limit = { exists = scope:secondary_recipient }
			is_clerical_appointment_valid_trigger = {
				TITLE = scope:target
				CANDIDATE = scope:secondary_recipient
				TOP_LIEGE = scope:actor
			}
		}
		scope:secondary_recipient = {
			custom_tooltip = {
				text = is_candidate_for_title_tt
				scope:target = {
					holder != scope:secondary_recipient
				}
				"appointment_candidate_accumulated_score(scope:target)" > 0
			}
		}
	}

	is_available = {
		can_influence_clerical_appointment_trigger = yes
		has_pam_dlc_trigger = yes
		piety >= massive_piety_value		#prev major_piety_value
	}

	is_shown = {
		scope:secondary_recipient ?= {
			# Without this a Head of Rite backing themselves would send the interaction to themselves
			this != faith.main_rite.head_of_rite
			is_clerical_appointment_valid_base_trigger = {
				CANDIDATE = scope:secondary_recipient
				TOP_LIEGE = scope:actor
			}
			NOR = {
				scope:actor.religious_head_or_challenger ?= this
				scope:secondary_recipient.religious_head_or_challenger ?= this
			}
		}
	}

	redirect = {
		scope:recipient = { save_scope_as = secondary_recipient }
		scope:actor.faith.main_rite.head_of_rite ?= { save_scope_as = recipient }
	}

	populate_recipient_list = {
		scope:actor = {
			if = {
				limit = {
					is_clerical_appointment_valid_base_trigger = {
						CANDIDATE = this
						TOP_LIEGE = scope:actor
					}
				}
				add_to_list = characters
			}
			if = {
				limit = {
					puppet:theological_agent_puppet ?= {
						is_clerical_appointment_valid_base_trigger = {
							CANDIDATE = this
							TOP_LIEGE = scope:actor
						}
					}
				}
				puppet:theological_agent_puppet = { add_to_list = characters }
			}
			if = {
				limit = {
					puppet:external_theological_agent_puppet ?= {
						is_clerical_appointment_valid_base_trigger = {
							CANDIDATE = this
							TOP_LIEGE = scope:actor
						}
					}
				}
				puppet:external_theological_agent_puppet = { add_to_list = characters }
			}
			every_relation = {
				type = friend
				limit = {
					is_clerical_appointment_valid_base_trigger = {
						CANDIDATE = this
						TOP_LIEGE = scope:actor
					}
				}
				add_to_list = characters
			}
			every_relation = {
				type = lover
				limit = {
					is_clerical_appointment_valid_base_trigger = {
						CANDIDATE = this
						TOP_LIEGE = scope:actor
					}
				}
				add_to_list = characters
			}
			#ADDED - try to promote family members too
			every_close_family_member = {
				limit = {
					is_clergy = yes
					is_clerical_appointment_valid_base_trigger = {
						CANDIDATE = this
						TOP_LIEGE = scope:actor
					}
				}
				add_to_list = characters
			}
		}
	}

	auto_accept = yes

	send_option = {
		flag = as_much_as_needed_candidacy_option
		localization = as_much_as_needed_candidacy_option_desc
	}
	send_option = {
		flag = major_candidacy_option
		localization = major_candidacy_option_desc
	}
	send_option = {
		flag = medium_candidacy_option
		localization = medium_candidacy_option_desc
	}

	cost = {
		piety = {
			value = {
				add = 0
				switch = {
					trigger = yes
					scope:as_much_as_needed_candidacy_option = {
						add = {
							value = pam_clerical_candidacy_outbid_piety_cost_value
							desc = BASE
						}
					}
					scope:major_candidacy_option = {
						add = {
							value = pam_clerical_candidacy_major_piety_cost_value
							desc = BASE
						}
					}
					scope:medium_candidacy_option = {
						add = {
							value = pam_clerical_candidacy_medium_piety_cost_value
							desc = BASE
						}
					}
				}
			}
		}
	}

	on_accept = {
		switch = {
			trigger = yes
			scope:as_much_as_needed_candidacy_option = {
				save_scope_value_as = {
					name = support_candidate_score
					value = {
						value = clerical_appointment_score_max_value
						multiply = scope:actor.appointment_score_interaction_multiplier_modifiers_value
						multiply = scope:actor.pam_theocratic_candidacy_situation_multiplier_value
					}
				}
			}
			scope:major_candidacy_option = {
				save_scope_value_as = {
					name = support_candidate_score
					value = {
						value = appointment_score_major_value
						multiply = scope:actor.appointment_score_interaction_multiplier_modifiers_value
						multiply = scope:actor.pam_theocratic_candidacy_situation_multiplier_value
					}
				}
			}
			scope:medium_candidacy_option = {
				save_scope_value_as = {
					name = support_candidate_score
					value = {
						value = appointment_score_medium_value
						multiply = scope:actor.appointment_score_interaction_multiplier_modifiers_value
						multiply = scope:actor.pam_theocratic_candidacy_situation_multiplier_value
					}
				}
			}
		}

		scope:target = {
			change_appointment_investment = {
				target = scope:secondary_recipient
				investor = scope:actor
				value = scope:support_candidate_score
			}
		}

		scope:secondary_recipient ?= {
			if = {
				limit = {
					this != scope:actor
					is_ai = no
				}
				hidden_effect = {
					send_interface_message = {
						type = msg_candidacy_improved
						title = supported_candidacy_toast
						left_icon = scope:actor
						right_icon = scope:target
						show_as_tooltip = {
							scope:target = {
								change_appointment_investment = {
									target = scope:secondary_recipient
									investor = scope:actor
									value = scope:support_candidate_score
								}
							}
						}
					}
				}
			}
		}

		# Christian Situation Catalysts
		if = {
			limit = {
				scope:actor = {
					NOR = {
						government_has_flag = government_is_ecclesiastical
						this = faith.main_rite.head_of_rite
					}
				}
				situation:the_christian_church ?= {
					situation_top_has_catalyst = catalyst_secular_influenced_clerical_candidacy
				}
			}
			situation:the_christian_church = {
				trigger_situation_catalyst = catalyst_secular_influenced_clerical_candidacy
			}
		}
		if = {
			limit = {
				scope:actor ?= scope:secondary_recipient
				scope:actor = {
					government_has_flag = government_is_ecclesiastical
				}
				situation:the_christian_church ?= {
					situation_top_has_catalyst = catalyst_clergy_influenced_own_candidacy
				}
			}
			situation:the_christian_church = {
				trigger_situation_catalyst = catalyst_clergy_influenced_own_candidacy
			}
		}
	}

	ai_targets = {
		ai_recipients = self
	}
	ai_targets = {
		ai_recipients = puppets
	}
	ai_targets = {
		ai_recipients = scripted_relations
		max = 20
	}

	# A crown returns to its capital see readily; a count has one see and rarely the piety to contest it
	ai_frequency_by_tier = {
		barony = 0
		county = 240
		duchy = 60
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	# Score the piety tiers explicitly so the AI does not fall back on send_option declaration order
	ai_will_do = {
		base = 0

		modifier = {
			scope:medium_candidacy_option = yes
			piety >= pam_clerical_candidacy_medium_piety_cost_value
			add = 100
		}
		modifier = {
			scope:major_candidacy_option = yes
			piety >= pam_clerical_candidacy_major_piety_cost_value
			add = 200
		}
		modifier = {
			scope:as_much_as_needed_candidacy_option = yes
			piety >= pam_clerical_candidacy_outbid_piety_cost_value
			add = 300
		}

		modifier = {
			exists = scope:secondary_recipient
			has_relation_rival = scope:secondary_recipient
			factor = 0
		}
		
		#ADDED - those 2
		modifier = {
			piety <= monumental_piety_value
			factor = 0.25
		}
		modifier = {
			exists = scope:secondary_recipient
			"opinion(scope:secondary_recipient)" < 0
			factor = 0
		}
	}
