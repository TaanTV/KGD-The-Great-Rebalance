
local_arbitration_perk = {
	lifestyle = wanderer_lifestyle

	position = { 1 3 }
	icon = node_wanderer

	government_character_modifier = {
		flag = government_is_settled

		flag = government_is_nomadic
		epidemic_resistance = 5
		monthly_control_change_when_below_baseline_add = 0.08
	}

	name = {
		first_valid = {

				trigger = { government_has_flag = government_is_merchant }

			}
			triggered_desc = {
				trigger = { has_government = landless_adventurer_government }

				trigger = { government_has_flag = government_is_nomadic }

		}
	}

	effect = {
		if = {
			limit = {
				government_has_flag = government_is_merchant
			}
			custom_description_no_bullet = {
				text = local_arbitration_perk_merchant_effect
			}
		}
		else_if = {
			limit = {
				government_has_flag = government_is_settled
			}
			custom_description_no_bullet = {

			}
		}
		else_if = {

				government_has_flag = government_is_landless_adventurer

				text = local_arbitration_perk_2_effect
			}

			switch = {

				camp_purpose_wanderers = { custom_description_no_bullet = { text = local_arbitration_wanderers_purpose } }

				camp_purpose_legitimists = { custom_description_no_bullet = { text = local_arbitration_legitimists_purpose } }
