
### LEARNING - SCHOLARSHIP PERKS

# Planned Cultivation
planned_cultivation_perk = {
	lifestyle = learning_lifestyle
	tree = scholarship
	position = { 2 1 }
	icon = node_learning
	
	parent = scientific_perk
	
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { has_government = landless_adventurer_government }
				desc = planned_cultivation_perk_adventurer_name
			}
			desc = planned_cultivation_perk_name
		}
	}
	
	government_character_modifier = {
		flag = government_is_landless_adventurer
		trait_track_travel_xp_gain_mult = 0.5	#prev 1
		men_at_arms_recruitment_cost = -0.1		#prev -0.25
	}
	
	government_character_modifier = {
		flag = government_is_nomadic
		county_fertility_growth_mult = 0.15
	}
	
	effect = {
		if = {
			limit = {
				NOR = {
					government_has_flag = government_is_landless_adventurer
					government_has_flag = government_is_nomadic
				}
			}
			custom_description_no_bullet = {
				text = planned_cultivation_perk_effect
			}
		}
		if = {
			limit = {
				government_has_flag = government_is_landless_adventurer
			}
			custom_tooltip = {
				text = adventurer_planned_cultivation_perk_effect
			}
		}
	}
}

# Sanctioned Loopholes
sanctioned_loopholes_perk = {
	lifestyle = learning_lifestyle
	tree = scholarship
	position = { 1 4 }
	icon = node_learning
	
	parent = learn_on_the_job_perk
	
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { has_government = landless_adventurer_government }
				desc = sanctioned_loopholes_perk_adventurer_name
			}
			triggered_desc = {
				trigger = {
					OR = {
						government_has_flag = government_has_merit
						government_has_flag = government_is_japan_administrative
						government_has_flag = government_is_japan_feudal
					}
				}
				desc = scholar_gentleman
			}
			desc = sanctioned_loopholes_perk_name
		}
	}
	
	government_character_modifier = {
		flag = government_is_landless_adventurer
		artifact_decay_reduction_mult = -0.5	#prev -0.75
	}
	government_character_modifier = {
		flag = government_has_merit
		confucian_education_xp_gain_mult = 0.15
		monthly_merit = 2
	}
	government_character_modifier = {
		flag = government_is_japan_administrative
		confucian_education_xp_gain_mult = 0.15
		domicile_monthly_piety_add = 0.5
	}
	government_character_modifier = {
		flag = government_is_japan_feudal
		confucian_education_xp_gain_mult = 0.15
		domicile_monthly_piety_add = 0.5
	}
	
	
	effect = {
		if = {
			limit = {
				NOR = {
					government_has_flag = government_is_landless_adventurer
					government_has_flag = government_has_merit
					government_has_flag = government_is_japan_administrative
					government_has_flag = government_is_japan_feudal
					pam_beseech_claims_access_trigger = yes
				}
			}
			custom_description_no_bullet = {
				text = sanctioned_loopholes_perk_effect
			}
		}
		if = {
			limit = {
				pam_beseech_claims_access_trigger = yes
			}
			custom_description_no_bullet = {
				text = sanctioned_loopholes_perk_effect_beseech_claim
			}
		}
		if = {
			limit = {
				has_pam_dlc_trigger = yes
				government_has_flag = government_is_theocracy
				pam_is_archbishop_or_higher_trigger = yes
			}
			custom_description_no_bullet = {
				text = sanctioned_loopholes_perk_effect_sell_claim
			}
		}
		if = {
			limit = {
				government_has_flag = government_is_landless_adventurer
			}	
			custom_tooltip = {
				text = adventurer_sanctioned_loopholes_perk_effect
			}
		}
	}
}
