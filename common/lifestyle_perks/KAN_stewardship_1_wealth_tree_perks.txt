
### STEWARDSHIP - WEALTH PERKS

# Heregeld
heregeld_perk = {
	lifestyle = stewardship_lifestyle
	tree = wealth
	position = { 0 1.25 }
	icon = node_stewardship
	
	parent = golden_obligations_perk
	
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { has_government = landless_adventurer_government }
				desc = heregeld_perk_adventurer_name
			}
			desc = heregeld_perk_name
		}
	}
	
	government_character_modifier = {
		flag = government_is_landless_adventurer
		invert_check = yes
		vassal_tax_contribution_mult = 0.1
		tributary_tax_mult = 0.1
	}
	
	government_character_modifier = {
		flag = government_is_landless_adventurer
		provisions_capacity_mult = 0.33		#prev 0.5
	}
}


# Detailed Ledgers
detailed_ledgers_perk = {
	lifestyle = stewardship_lifestyle
	tree = wealth
	position = { 0 2.5 }
	icon = node_stewardship
	
	parent = heregeld_perk
	
	name = {
		first_valid = {
			triggered_desc = {
				trigger = { has_government = landless_adventurer_government }
				desc = detailed_ledgers_perk_adventurer_name
			}
			desc = detailed_ledgers_perk_name
		}
	}
	
	government_character_modifier = {
		flag = government_is_landless_adventurer
		invert_check = yes
		republic_government_opinion = 20
		republic_government_tax_contribution_mult = 0.15
	}
	
	government_character_modifier = {
		flag = government_is_landless_adventurer
		provisions_loss_mult = -0.1		#prev -0.2
	}
	
	government_character_modifier = {
		flag = government_is_nomadic
		monthly_income_from_herd_mult = 0.1
	}
	
	government_character_modifier = {
		flag = government_is_administrative
		administrative_government_tax_contribution_mult = 0.1
	}

	government_character_modifier = {
		flag = government_has_treasury
		monthly_treasury_from_salary_budget_mult = 0.1
	}

	#Ecclesiastical governments have a treasury but no salary budget, so the block above does nothing for them
	government_character_modifier = {
		flag = government_is_ecclesiastical
		domicile_monthly_gold_add = 1
	}

	effect = {
		if = {
			limit = { government_has_flag = government_is_mandala }	
			custom_description_no_bullet = { text = mandala_detailed_ledgers_perk_effect }
		}
	}
}
