underage_diarch_loyalty_restriction = {
	trigger_if = {
		limit = { 
			OR = {
				age >= 16
				has_active_diarchy = no
				NOT = { has_game_rule = bkt_additional_content_enabled }
			}
		}
	}
	trigger_else_if = { 
		limit = { age < 4 has_active_diarchy = yes }
		diarch = { save_temporary_scope_as = diarch_scope }
		custom_tooltip = {
			text = diarch_is_not_loyal_tt
			diarch.diarch_loyalty >= diarch_loyalty_visibly_loyal_threshold
		}
	}
	trigger_else_if = {
		limit = { age < 16 has_active_diarchy = yes }
		diarch = { save_temporary_scope_as = diarch_scope }
		custom_tooltip = {
			text = diarch_is_disloyal_tt
			diarch.diarch_loyalty > diarch_loyalty_visibly_disloyal_threshold
		}
	}
	trigger_else = { }
}

puppet_or_actor_doesnt_have_troops_in_recipient_lands = {
	custom_tooltip = {
		text = you_have_troops_in_their_subrealm
		scope:actor = {
			is_ai = no	#i doubt AI cheeses it ever
			OR = {
				scope:recipient.highest_held_title_tier <= tier_barony #those dont really matter for this check, its cheesable anyway
				NOT = {
					any_army = {
						location.county.holder ?= {
							target_is_same_character_or_above = scope:recipient
						}
					}
				}
			}
		}
	}
}

this_is_real_ally_of_target = {
	$TARGET$ = { save_temporary_scope_as = target }
	NOR = {
		highest_held_title_tier <= tier_barony
		government_has_flag = government_is_mercenary
		trigger_if = {
			limit = {
				NAND = {
					government_has_flag = can_call_vassals_into_wars
					scope:target = { government_has_flag = can_call_vassals_into_wars }
				}
			}	
			OR = {
				target_is_liege_or_above = scope:target
				target_is_vassal_or_below = scope:target
			}
		}
	}
}
