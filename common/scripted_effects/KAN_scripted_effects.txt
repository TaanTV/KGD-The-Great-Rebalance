create_same_type_special_troop = {
	# Has $MAA_TYPE$
	#save_temporary_scope_value_as = { name = original_troop_stack_size value = maa_size }
	save_temporary_scope_value_as = { name = original_troop_men_count value = maa_current_troops_count }
	if = {
		limit = { has_game_rule = kan_on_landed_army_keep_as_special_troops_inheritable }
		scope:scoped_adventurer = {
			spawn_army = {
				name = event_troop_default_name
				men_at_arms = { # optional, multiple can be specified. Need either levies or MAA
				   type = $MAA_TYPE$
				   #stacks = scope:original_troop_stack_size
				   men = scope:original_troop_men_count
				}
				location = scope:scoped_adventurer.location
				inheritable = yes
			}
		}
	}
	else = {
		scope:scoped_adventurer = {
			spawn_army = {
				name = event_troop_default_name
				men_at_arms = { # optional, multiple can be specified. Need either levies or MAA
				   type = $MAA_TYPE$
				   #stacks = scope:original_troop_stack_size
				   men = scope:original_troop_men_count
				}
				location = scope:scoped_adventurer.location
				inheritable = no
			}
		}
	}
}

create_250_levies_per_unit_special_troop = {
	#save_temporary_scope_value_as = { name = original_troop_stack_size value = maa_size }
	save_temporary_scope_value_as = { 
		name = levy_size_multiplier 
		value = {
			value = maa_size 
			multiply = {
				value = maa_current_troops_count
				divide = { value = maa_max_troops_count min = 1 }
			}
		}
	}
	if = {
		limit = { has_game_rule = kan_on_landed_army_keep_as_special_troops_inheritable }
		scope:scoped_adventurer = {
			spawn_army = {
				name = event_troop_default_name
				levies = {
					value = 250
					multiply = scope:levy_size_multiplier
					round_to = 5
				}
				location = scope:scoped_adventurer.location
				inheritable = yes 
			}
		}
	}
	else = {
		scope:scoped_adventurer = {
			spawn_army = {
				name = event_troop_default_name
				levies = {
					value = 250
					multiply = scope:levy_size_multiplier
					round_to = 5
				}
				location = scope:scoped_adventurer.location
				inheritable = no
			}
		}
	}
}

convert_maa_to_special_troops = {
	if = {
		limit = {
			NOT = { has_game_rule = kan_on_landed_army_keep_as_men_at_arms }
		}
		save_scope_as = scoped_adventurer
		every_maa_regiment = {
			if = {
				limit = { NOT = { is_unit_type = siege_weapon } }
				convert_this_maa_into_special_troop = yes
				destroy_maa_regiment = yes
			}
		}
	}
}
