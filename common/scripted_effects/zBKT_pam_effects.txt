pam_ai_personal_tenet_pulse_effect = {
	if = {
		limit = {
			is_physically_able_ai = yes
		}

		# Phase 0: cardinals may hold beliefs their rite prohibits, so a new pontiff can arrive carrying one
		# Ahead of Phase 1 so the freed slot refills this pulse; on_personal_tenet_loss clears the secret
		if = {
			limit = {
				pam_holds_college_elected_church_seat_trigger = yes
				pam_has_forbidden_personal_tenet_trigger = yes
			}
			faith.main_rite = { save_temporary_scope_as = pam_pontiff_rite }
			while = {
				count = 8
				limit = { pam_has_forbidden_personal_tenet_trigger = yes }
				random_personal_tenet = {
					limit = {
						scope:pam_pontiff_rite = {
							has_tenet_status = {
								tenet = prev
								status = prohibited
							}
						}
					}
					root = { remove_personal_tenet = prev }
				}
			}
		}

		# Phase 1: fill every empty personal tenet slot. No chance gate - empty slots are wasted SF and immersion.
		# 8 = safety upper bound on slot cap (base 1 + up to ~5 from piety levels, with headroom).
		while = {
			count = 8
			limit = {
				free_personal_tenet_slots > 0
				any_desired_tenet = {
					only_positive = yes
					threshold = 1
					selection_count = 15	#prev 97
					pam_ai_personal_tenet_candidate_trigger = yes
				}
			}

			random_desired_tenet = {
				only_positive = yes
				threshold = 1
				selection_count = 15	#prev 97
				limit = { pam_ai_personal_tenet_candidate_trigger = yes }
				weight = {
					base = 10
					modifier = {
						factor = root.anathema_to_faith_personal_tenet_weight_factor
						tenet_is_anathema_to_faith_trigger = { FAITH = root.faith }
					}
					modifier = {
						factor = root.intrinsic_to_faith_personal_tenet_weight_factor
						tenet_is_intrinsic_to_faith_trigger = { FAITH = root.faith }
					}
				}
				root = { set_personal_tenet = prev }
			}
		}

		# Nothing known would raise their fulfillment: fill the rest with harmless core tenets of their own rite, as at game start
		while = {
			count = 8
			limit = {
				free_personal_tenet_slots > 0
				rite = {
					any_rite_tenet = { pam_ai_personal_tenet_fallback_trigger = yes }
				}
			}

			rite = {
				random_rite_tenet = {
					limit = { pam_ai_personal_tenet_fallback_trigger = yes }
					root = { set_personal_tenet = prev }
				}
			}
		}

		# Phase 2: chance-gated swap, only when there are no free slots left to fill.
		# Only a tenet that gives no fulfillment is swapped out.
		if = {
			limit = {
				free_personal_tenet_slots <= 0
				any_personal_tenet = { pam_personal_tenet_gives_no_fulfillment_trigger = yes }
				any_desired_tenet = {
					only_positive = yes
					threshold = 1
					selection_count = 3		#prev 97
					pam_ai_personal_tenet_candidate_trigger = yes
				}
			}

			random = {
				chance = 30
				modifier = {
					add = 15
					has_trait = zealous
				}
				modifier = {
					add = -15
					has_trait = cynical
				}
				modifier = {
					add = {
						value = ai_zeal
						divide = 5
					}
				}
				modifier = {
					add = {
						value = ai_boldness
						divide = 5
					}
				}

				# Pick the new tenet - Random of any positive-SF known tenet, not strictly the best.	#ADDED - nah, strictly the best ones
				random_desired_tenet = {
					only_positive = yes
					threshold = 1
					selection_count = 3		#prev 97
					limit = { pam_ai_personal_tenet_candidate_trigger = yes }
					weight = {
						base = 10
						modifier = {
							factor = root.anathema_to_faith_personal_tenet_weight_factor
							tenet_is_anathema_to_faith_trigger = { FAITH = root.faith }
						}
						modifier = {
							factor = root.intrinsic_to_faith_personal_tenet_weight_factor
							tenet_is_intrinsic_to_faith_trigger = { FAITH = root.faith }
						}
					}
					save_temporary_scope_as = pam_personal_tenet_replacement
					root = {
						random_personal_tenet = {
							limit = { pam_personal_tenet_gives_no_fulfillment_trigger = yes }
							root = {
								remove_personal_tenet = prev
								set_personal_tenet = scope:pam_personal_tenet_replacement
							}
						}
					}
					root = { force_set_personal_tenet = prev }
				}
			}
		}
	}
}

pam_levy_the_tithes_decision_effects = {
	every_vassal = {
		limit = {
			OR = {
				government_has_flag = government_is_theocracy
				government_has_flag = government_is_ecclesiastical
			}
		}
		custom = every_theocratic_vassal_tt
		send_interface_message = {
			type = event_gold_bad
			title = pam_levy_the_tithes_decision
			left_icon = root
			pay_short_term_gold = {
				target = root
				gold = minor_gold_value
			}
			add_opinion = {
				modifier = tithed_opinion
				target = root
				opinion = -30	#prev -25
			}
		}
		if = { 
			limit = { faith = prev.faith }
			save_scope_value_as = {
				name = same_faith_vassals_affected
				value = 1
			}
		}
	}
	if = {
		limit = { 
			exists = cp:councillor_court_chaplain 
			exists = scope:same_faith_vassals_affected
		}
		cp:councillor_court_chaplain = {
			progress_towards_rival_effect = {
				CHARACTER = root
				REASON = rival_levied_tithes
				OPINION = default_rival_opinion
			}
		}
	}
	#ADDED - zealots also dont like u for that
	if = {
		limit = { exists = scope:same_faith_vassals_affected }
		every_vassal = {
			vassal_stance = zealot
			limit = {
				NOR = {
					government_has_flag = government_is_theocracy
					government_has_flag = government_is_ecclesiastical
				}
			}
			custom = every_zealot_vassal
			add_opinion = {
				target = prev
				modifier = annoyed_opinion
				opinion = -20
			}
		}
		save_scope_as = tither
		faith ?= {
			change_fervor = {
				value = {
					value = trivial_fervor_loss
					if = {
						limit = { prev.highest_held_title_tier >= tier_kingdom }
						value = slight_fervor_loss
					}
				}
				desc = fervor_loss_ruler_tithes_levied
			}
		}
	}
	else = {
		every_vassal = {
			vassal_stance = zealot
			limit = {
				NOR = {
					government_has_flag = government_is_theocracy
					government_has_flag = government_is_ecclesiastical
				}
			}
			custom = every_zealot_vassal
			add_opinion = {
				target = prev
				modifier = suspicion_opinion
				opinion = -10
			}
		}
	}
}
