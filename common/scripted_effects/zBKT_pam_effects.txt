pam_levy_the_tithes_decision_effects = {
	every_vassal = {
		limit = {
			OR = {
				government_has_flag = government_is_theocracy
				government_has_flag = government_is_ecclesiastical
			}
		}
		custom = every_theocratic_vassal_tt
		#ADDED - send a notification at least
		send_interface_message = {
			type = send_interface_message_bad
			title = pam_levy_the_tithes_decision
			left_icon = this
			right_icon = prev
		pay_short_term_gold = {
			target = root
			gold = minor_gold_value
		}
			#ADDED - they should be at least mildly annoyed, no?
			add_opinion = {
				target = prev
				modifier = annoyed_opinion
				opinion = -30
			}
		}
		if = { 
			limit = { faith = prev.faith }
			save_scope_value_as = {
				name = same_faith_vassals_affected
				value = 1
			}
		}
	}
	if = {
		limit = { 
			exists = cp:councillor_court_chaplain 
			exists = scope:same_faith_vassals_affected
		}
		cp:councillor_court_chaplain = {
			progress_towards_rival_effect = {
				CHARACTER = root
				REASON = rival_levied_tithes
				OPINION = default_rival_opinion
			}
		}
	}
	#ADDED - zealots also dont like u for that
	if = {
		limit = { exists = scope:same_faith_vassals_affected }
		every_vassal = {
			vassal_stance = zealot
			limit = {
				NOR = {
					government_has_flag = government_is_theocracy
					government_has_flag = government_is_ecclesiastical
				}
			}
			custom = every_zealot_vassal
			add_opinion = {
				target = prev
				modifier = annoyed_opinion
				opinion = -25
			}
		}
		save_scope_as = tither
		faith ?= {
			change_fervor = {
				value = {
					value = trivial_fervor_loss
					if = {
						limit = { prev.highest_held_title_tier >= tier_kingdom }
						value = slight_fervor_loss
					}
				}
				desc = fervor_loss_ruler_tithes_levied
			}
		}
	}
	else = {
		every_vassal = {
			vassal_stance = zealot
			limit = {
				NOR = {
					government_has_flag = government_is_theocracy
					government_has_flag = government_is_ecclesiastical
				}
			}
			custom = every_zealot_vassal
			add_opinion = {
				target = prev
				modifier = suspicion_opinion
				opinion = -10
			}
		}
	}
}
