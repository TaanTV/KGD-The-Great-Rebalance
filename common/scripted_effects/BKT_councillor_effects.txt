got_council_position_effect = {
	save_scope_as = new_councillor
	# Travel Leaders can't be councillors, but should appear in the councillor list
	if = {
		limit = { has_court_position = travel_leader_court_position }
		liege = {
			revoke_court_position = {
				recipient = scope:new_councillor
				court_position = travel_leader_court_position
			}
		}
	}
	#If there is a fired opinion remove it
	if = {
		limit = {
			#ADDED
			exists = liege
			has_opinion_modifier = {
				modifier = fired_from_council_opinion
				target = liege
			}			
		}
		remove_opinion = {
			modifier = fired_from_council_opinion
			target = liege
		}
	}
	# No longer angry for losing court position - this is *better*!
	if = {
		limit = {
			#ADDED
			exists = liege
			has_opinion_modifier = {
				modifier = revoked_court_position_opinion
				target = liege
			}			
		}
		remove_opinion = {
			modifier = revoked_court_position_opinion
			target = liege
		}
	}

	add_character_flag = is_in_the_council
	if = {
		limit = {
			has_pam_dlc_trigger = yes
			NOT = { religion = religion:christianity_religion }
			scope:councillor_liege ?= {
				highest_held_title_tier >= tier_kingdom
				any_character_situation = {
					situation_type = the_christian_church
					situation_top_has_catalyst = catalyst_the_christian_church_ruler_hired_non_christian_councillor
				}
			}
		}
		situation:the_christian_church = {
			trigger_situation_catalyst = catalyst_the_christian_church_ruler_hired_non_christian_councillor
		}
	}
	
	#ADDED
	if = {
		limit = { 
			#is_ruler = yes
			liege ?= { is_ai = no }	#AI can wait for the scheduled recalc, it's fine
		}
		liege = { recalc_vassal_stance_council_bonuses = yes }
	}
}
