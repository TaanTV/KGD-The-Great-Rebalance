# Levy the Tithes
pam_levy_the_tithes_decision = {
	picture = {
		trigger = { culture = { has_graphical_east_asia_culture_group_trigger = yes } }
		reference = "gfx/interface/illustrations/decisions/tgp_scholars.dds"
	}
	decision_group_type = religious
	picture = {
		trigger = { culture = { has_graphical_mena_culture_group_trigger = yes } }
		reference = "gfx/interface/illustrations/decisions/fp3_decision_tax_collector.dds"
	}
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_spend_money.dds"
	}

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		#ADDED - OR and religion check
		OR = {
		any_character_situation = { # You are part of the Christian Situation
			situation_type = the_christian_church
		}
		faith.religion != religion:christianity_religion	
		}
		has_pam_dlc_trigger = yes
		is_clergy = no
		is_ruler = yes
		#ADDED - dont show this to ones where it would be weird
		NOR = {
			government_has_flag = government_is_in_steppe
			government_has_flag = government_is_mandala
			rite = { rite_has_parameter = unreformed }
			AND = {
				rite = { rite_has_parameter = lay_clergy }
				NOT = { any_vassal = { government_has_flag = government_is_theocracy } }
			}
		}
	}

	is_valid = {
		#ADDED
		trigger_if = {
			limit = { faith.religion = religion:christianity_religion }
		custom_tooltip = {
			text = pam_christian_church_instability_tt
			top_participant_group:the_christian_church ?= {
				has_participant_group_parameter = the_christian_church_levy_the_tithes_decision
			}
		}
		}
		any_vassal = {
			custom_tooltip = {
				text = pam_levy_the_tithes_decision_theocracy_tt
				government_has_flag = government_is_theocracy
			}
		}
	}

	cooldown = { years = 10 }

	cost = {
		piety = {
			value = massive_piety_value		#prev major_piety_value
			if = {
				limit = {
					NAND = {
						is_ai = no			#minor premature optimisation and let AI breathe a bit
						any_vassal = { government_has_flag = government_is_theocracy faith = prev.faith }
					}
				}
				value = medium_piety_value
			}
			multiply = spiritual_fulfillment_tithe_legate_cost_mult
		}
	}

	effect = {
		show_as_tooltip = {
			pam_levy_the_tithes_decision_effects = yes
		}
		if = {
			limit = {
				NOT = { any_vassal = { government_has_flag = government_is_theocracy } }
			}
			custom_tooltip = pam_levy_the_tithes_decision_effects_no_vassals_tt
		}
		trigger_event = pam_decision_events.0008
	}

	ai_potential = {
		ai_greed > low_negative_ai_value	#prev medium_positive_ai_value
		#ADDED - this and an OR
		trigger_if = {
			limit = { NAND = { rite = { rite_has_parameter = lay_clergy } faith = { has_doctrine_parameter = lay_clergy } } }
			ai_zeal < low_positive_ai_value
		}
		OR = {
			is_in_debt = yes
			spiritual_fulfillment < spiritual_fulfillment_minus_1
			AND = {			#since they probably cant have same faith vassals.... right?
				rite = { rite_has_parameter = lay_clergy }
				faith = { has_doctrine_parameter = lay_clergy }		
			}
			AND = {
				ai_greed > medium_positive_ai_value
				ai_zeal < very_low_negative_ai_value
			}
		}
		NOR = {
			has_trait = content
			has_trait = craven
		}
	}

	ai_will_do = {
		base = 10

		modifier = {
			add = 25
			has_trait = greedy
		}
		
		#ADDED
		modifier = {	#dooming
			spiritual_fulfillment < spiritual_fulfillment_minus_2
			add = 15
		}
		modifier = {	#extorting other-faith clergy is pretty cool
			rite = { rite_has_parameter = lay_clergy }
			faith = { has_doctrine_parameter = lay_clergy }
			ai_zeal > low_positive_ai_value
			add = 10
		}
		modifier = {	#sry need money
			is_in_debt = yes
			add = 40
		}
	}
}


pam_declare_rite_heresy_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_story/pam_the_christian_church_instability.dds"
	}
	decision_group_type = major

	desc = pam_declare_rite_heresy_decision_desc
	selection_tooltip = pam_declare_rite_heresy_decision_tooltip
	confirm_text = pam_declare_rite_heresy_decision_confirm

	sort_order = 108

	ai_check_interval_by_tier = {
		barony = 0
		county = 72
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		this = faith.religious_head
		faith = { has_papacy_trigger = no } # The pope uses bulls
		faith ?= {
			any_faith_rite = {
				pam_rite_can_be_declared_heresy_trigger = yes
			}
		}
	}

	is_valid = {
		trigger_if = {
			limit = { has_trait = zealous }
			custom_tooltip = {
				text = pam_declare_rite_heresy_decision_valid_zealous_tt
				faith = {
					any_faith_rite = {
						pam_rite_can_be_declared_heresy_trigger = yes
					}
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = pam_declare_rite_heresy_decision_valid_tt
				faith = {
					any_faith_rite = {
						pam_rite_can_be_declared_heresy_trigger = yes
					}
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_imprisoned = no
		custom_tooltip = {
			text = pam_rite_heresy_select_tt
			exists = scope:selected_item
		}
	}

	cooldown = { years = 5 }

	cost = {
		piety = major_piety_value
	}

	effect = {
		save_scope_as = bull_issuer
		scope:selected_item = { save_scope_as = heresy_rite }
		send_interface_toast = {
			title = pam_declare_rite_heresy_effect_toast_tt
			right_icon = scope:heresy_rite
			pam_declare_rite_heresy_effect = yes
		}
		faith = {
			every_faith_character = {
				limit = {
					is_ruler = yes
					this != scope:bull_issuer
				}
				trigger_event = { id = pam_decision_events.0012 days = 1 }
			}
		}
		if = {
			limit = { exists = scope:heresy_rite }
			hidden_effect = {
				if = {
					limit = {
						scope:heresy_rite.faith.religious_head ?= {
							NOT = { has_trait = heresiarch }
						}
					}
					scope:heresy_rite.faith.religious_head = { add_trait = heresiarch }
				}
				else_if = {
					limit = {
						scope:heresy_rite.main_rite.head_of_rite ?= {
							NOT = { has_trait = heresiarch }
						}
					}
					scope:heresy_rite.main_rite.head_of_rite = { add_trait = heresiarch }
				}
				if = {
					limit = {
						scope:heresy_rite.faith = { has_doctrine = special_doctrine_ecumenical_christian }
					}
					scope:heresy_rite.faith = { remove_doctrine = special_doctrine_ecumenical_christian }
				}
				if = {
					limit = {
						scope:heresy_rite.faith = { has_doctrine = special_doctrine_ecumenical_miaphysite }
					}
					scope:heresy_rite.faith = { remove_doctrine = special_doctrine_ecumenical_miaphysite }
				}
			}
		}
	}

	widget = {
		gui = "decision_view_widget_select_rite"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT"

		setup_items = {
			scope:actor = {
				faith = {
					every_faith_rite = {
						limit = {
							pam_rite_can_be_declared_heresy_trigger = yes
						}
						add_to_list = item_list
					}
				}
			}
		}

		default_item = {
			scope:actor = {
				faith = {
					random_faith_rite = {
						limit = {
							pam_rite_can_be_declared_heresy_trigger = yes
						}
						save_scope_as = selected_item
					}
				}
			}
		}

		is_item_valid = {
			pam_rite_can_be_declared_heresy_trigger = yes
		}

		ai_select_item = {
			scope:actor = {
				faith = {
					ordered_faith_rite = {
						limit = {
							pam_rite_can_be_declared_heresy_trigger = yes
						}
						order_by = "divergence(faith.main_rite)"
						save_scope_as = selected_item
					}
				}
			}
		}
	}

	ai_potential = {
		this = faith.religious_head
		faith = {
			any_faith_rite = {
				pam_rite_can_be_declared_heresy_trigger = yes
			}
		}
	}

	ai_will_do = {
		base = 30		#prev 20
		modifier = {
			add = 25
			has_trait = zealous
		}
		modifier = {
			add = 15
			has_trait = theologian
		}
		modifier = {
			add = -25
			has_trait = cynical
		}
		modifier = {
			add = -50
			rite = { rite_has_doctrine = doctrine_pluralism_pluralistic }
		}
		modifier = {
			add = -1000		#prev -100
			scope:selected_item = scope:actor.rite
		}
		
		#ADDED 
		modifier = {
			add = 15
			rite = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
		}
		modifier = {
			faith.fervor < 50
			add = {
				value = 70	#this being more than 50 is intentional
				subtract = faith.fervor
			}
		}
		modifier = {
			exists = puppeteer
			scope:selected_item ?= puppeteer.rite
			factor = 0
		}
	}
}

# Head of Faith: formally take a stance on a tenet known to the faith or learned personally.
pam_permit_tenet_decision = {

	#ADDED - title!
	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOR = {
						faith.religious_head ?= this
						AND = {
							OR = {
								government_has_flag = government_is_theocracy
								government_has_flag = government_is_mandala
							}
							rite.head_of_rite = this
							faith = { 
								has_doctrine_parameter = no_head_of_faith
								OR = {
									main_rite = root.rite
									has_doctrine_parameter = unreformed
								}
							}
						}
					}
				}
				desc = pam_permit_tenet_decision_via_puppet
			}
			desc = pam_permit_tenet_decision
		}
	}
	picture = {
		reference = "gfx/interface/illustrations/event_story/pam_the_christian_church_stability.dds"
	}
	decision_group_type = major

	desc = pam_permit_tenet_decision_desc
	selection_tooltip = pam_permit_tenet_decision_tooltip
	confirm_text = pam_permit_tenet_decision_confirm

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		#this = faith.religious_head
		faith = { has_papacy_trigger = no } # The pope uses bulls
		#ADDED - can be pressed if you are the puppeteer too, or if you're head of rite and there's no HOF
		trigger_if = {
			limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
			faith = {
				OR = {
					main_rite = root.rite
					has_doctrine_parameter = unreformed
				}
				trigger_if = {
					limit = { this = faith:christian_faith }
					has_global_variable = pam_great_schism_prevented
				}
			}
			this = rite.head_of_rite
			trigger_if = {
				limit = { is_clergy = no NOT = { government_has_flag = government_is_mandala } }
				puppet:theological_agent_puppet ?= { rite = prev.rite }
			}
		}
		trigger_else = {
			exists = faith.religious_head
			OR = {
				this = faith.religious_head
				faith.religious_head.puppeteer ?= this
			}
		}
	}
	
	#ADDED - this just looks better
	is_valid = {
		trigger_if = {
			limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
			OR = {
				custom_tooltip = {
					text = manage_holy_sites_realm_priest_puppet_tt
					# puppet: errors on non-playable characters, so we only check playable ones and tooltips never run it
					trigger_if = {
						limit = { is_playable_character = yes }
						exists = puppet:theological_agent_puppet
					}
					trigger_else = {
						always = no
					}
				}
				government_has_flag = government_is_theocracy
				government_has_flag = government_is_mandala
			}
			trigger_if = {
				limit = { NOR = { government_has_flag = government_is_theocracy government_has_flag = government_is_mandala } }
				puppet:theological_agent_puppet = { piety >= major_piety_value } 
			}
			custom_tooltip = {
				text = pam_shape_the_christian_church_decision_hor_tt
				always = yes		#already checked in is_shown anyway
			}
			custom_tooltip = {
				text = manage_holy_sites_piety_level_tt
				piety_level >= 4
			}
		}
		trigger_else_if = {
			limit = { this != faith.religious_head }
			faith.religious_head = { 
				save_temporary_scope_as = faith_authority
				piety >= major_piety_value 
			}
			custom_tooltip = {
				text = puppeting_faith_authority_tt
				always = yes		#already checked in is_shown anyway
			}
		}
		trigger_else = { }
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		#ADDED - also check HOF or whatever if need be
		trigger_if = {
			limit = { 
				faith = { has_doctrine_parameter = no_head_of_faith } 
				NOR = { 
					government_has_flag = government_is_mandala 
					government_has_flag = government_is_theocracy
				} 
			}
			puppet:theological_agent_puppet = { is_available_adult = yes } 
		}
		trigger_else_if = {
			limit = { this != faith.religious_head }
			faith.religious_head = { is_available_adult = yes }
		}
		trigger_else = { }
		custom_tooltip = {
			text = pam_tenet_stance_select_tt
			exists = scope:selected_item
		}
		custom_tooltip = {
			text = ecumenical_council_already_ongoing_tt
			NOT = { faith = { has_variable = ecumenical_council_ongoing } }
		}
		#ADDED - faith wide cooldown coz F you
		faith = { save_temporary_scope_as = faith_scope }
		custom_tooltip = {
			text = tenet_permit_cooldown_tt
			NOT = { faith = { has_variable = tenet_permit_cooldown } }
		}
	}

	cooldown = { years = 5 }	#prev 3

	cost = {
		#piety = medium_piety_value
		#ADDED - diff cost if you're not the HOF
		piety = {
			value = major_piety_value
			if = {
				limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
				value = 3000
			}
			else_if = {
				limit = { this != faith.religious_head }
				value = 1500
			}
		}
	}

	effect = {
		#save_scope_as = pam_tenet_stance_issuer
		#ADDED choose the issuer
		if = {
			limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
			puppet:theological_agent_puppet ?= { save_scope_as = pam_tenet_stance_issuer }
			if = {
				limit = { OR = { government_has_flag = government_is_theocracy government_has_flag = government_is_mandala } }
				save_scope_as = pam_tenet_stance_issuer
			}
		}
		else_if = {
			limit = { this != faith.religious_head }
			faith.religious_head = { save_scope_as = pam_tenet_stance_issuer }
		}
		else = { save_scope_as = pam_tenet_stance_issuer }
		save_scope_as = pam_tenet_stance_permit
		scope:selected_item = { save_scope_as = pam_tenet_stance_tenet }
		#ADDED
		if = {
			limit = { scope:pam_tenet_stance_issuer != this }
			scope:pam_tenet_stance_issuer = { add_piety = major_piety_loss }
		}
		faith = {
			set_tenet_status = {
				tenet = scope:selected_item
				status = permitted
			}
			change_fervor = {
				#value = significant_fervor_loss
				#ADDED - worse fervor impact for hofless
				value = {
					value = significant_fervor_loss
					if = {
						limit = { has_doctrine_parameter = no_head_of_faith }
						value = substantial_fervor_loss
					}
				}
				desc = fervor_loss_pam_tenet_permitted
			}
			if = {
				limit = { has_tenet_popularity_trigger = yes }
				change_tenet_popularity = {
					target = scope:selected_item
					change = medium_tenet_popularity_gain
				}
			}
			pam_faith_cleanup_forbidden_personal_tenet_secrets_effect = yes
			every_faith_character = {
				limit = {
					is_ruler = yes
					this != scope:pam_tenet_stance_issuer
				}
				trigger_event = { id = pam_decision_events.0011 days = 1 }
			}
			#ADDED - faith-wide cooldown
			hidden_effect = {
				set_variable = {
					name = tenet_permit_cooldown
					years = 5
				}
			}
		}
	}

	widget = {
		gui = "decision_view_widget_select_tenet"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT"

		# List is tenets the HoF personally knows + tenets the rite knows
		setup_items = {
			scope:actor = {
				ordered_character_tenet = {
					knowledge_filter = known
					order_by = {
						value = pam_raw_tenet_popularity_value
						multiply = -1
					}
					max = 200
					check_range_bounds = no
					add_to_list = item_list
				}
			}
		}

		default_item = {
			scope:actor = {
				random_character_tenet = {
					knowledge_filter = known
					limit = { pam_tenet_can_be_permitted_trigger = yes }
					save_scope_as = selected_item
				}
			}
		}

		is_item_valid = {
			pam_tenet_can_be_permitted_trigger = yes
		}

		ai_select_item = {
			scope:actor = {
				if = { # A tenet the faith adores gets sanctioned whether or not the Head of Faith cares for it
					limit = {
						#ADDED - check this?????
						faith = { has_tenet_popularity_trigger = yes }
						any_character_tenet = {
							knowledge_filter = known
							pam_tenet_can_be_permitted_trigger = yes
							pam_tenet_is_overwhelmingly_popular_trigger = { CHARACTER = scope:actor }
						}
					}
					ordered_character_tenet = {
						knowledge_filter = known
						limit = {
							pam_tenet_can_be_permitted_trigger = yes
							pam_tenet_is_overwhelmingly_popular_trigger = { CHARACTER = scope:actor }
						}
						order_by = pam_raw_tenet_popularity_value
						position = 0
						save_scope_as = selected_item
					}
				}
				#ADDED - intercept for faiths with no tenet popularity
				else_if = {
					limit = {
						NOT = { faith = { has_tenet_popularity_trigger = yes } }
					}
					random_desired_tenet = {
						limit = {
							pam_tenet_can_be_permitted_trigger = yes
							pam_tenet_ai_may_permit_trigger = { CHARACTER = scope:actor }
						}
						only_positive = yes
						threshold = 1
						selection_count = 5
						weight = {
							base = 10
						}
						save_scope_as = selected_item
					}
				}
				else = {
					random_desired_tenet = {
						limit = {
							pam_tenet_can_be_permitted_trigger = yes
							pam_tenet_ai_may_permit_trigger = { CHARACTER = scope:actor }
						}
						only_positive = yes
						threshold = 1
						selection_count = 97
						weight = {
							base = 10
							modifier = {
								add = pam_tenet_popularity_impact_value
							}
						}
						save_scope_as = selected_item
					}
				}
			}
		}
	}

	ai_potential = {
		#this = faith.religious_head
		#ADDED - kind of changed the whole thing tbh
		trigger_if = {
			limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
			this = rite.head_of_rite
			puppet:theological_agent_puppet ?= { rite = prev.rite }
		}
		trigger_else = {
			exists = faith.religious_head
			OR = {
				this = faith.religious_head
				faith.religious_head.puppeteer ?= this
			}
		}
		trigger_if = {
			faith = { has_tenet_popularity_trigger = yes }
			OR = {
				any_desired_tenet = {
					only_positive = yes
					threshold = 1
					selection_count = 97
					root.rite = {
						NOR = {
							has_tenet_status = { tenet = prev status = permitted }
