elevate_holy_site_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_mentor_sp.dds"
	}
	decision_group_type = religious

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24 # Head of faiths are Duchy level titles
		kingdom = 24
		empire = 12
		hegemony = 12
	}

	# Opened via OpenGameViewData from the Holy Site window or theological puppet action
	is_invisible = yes

	is_valid_showing_failures_only = {
		exists = scope:selected_item
	}

	is_valid = {
		save_temporary_scope_as = root
		exists = scope:selected_item

		can_manage_faith_holy_sites_trigger = yes

		scope:selected_item ?= {
			root.faith = {
				is_faith_holy_site = prev
				NOT = { is_faith_eminent_holy_site = prev }
				save_temporary_scope_as = faith
				custom_tooltip = {
					text = elevate_holy_site_decision_under_eminent_holy_sites_max
					list_size:eminent_holy_site < root.faith.religion.eminent_holy_sites_max
				}
			}
			custom_tooltip = {
				text = elevate_holy_site_decision_target_in_realm_or_head_of_faith
				OR = {
					root.faith.religious_head ?= root
					holy_site_barony.county.holder = {	#ADDED - county holder coz barony holder may be missing
						target_is_same_character_or_above = root
					}
				}
			}
			custom_tooltip = {
				text = elevate_holy_site_decision_county_holder_correct_faith
				holy_site_county.holder.faith = root.faith
			}
		}

		# Show usage of free holy site action, if appropriate
		faith = {
			holy_site_free_action_show_usage_trigger = yes
		}
	}

	cost = {
		piety = {
			value = 1000
			if = {
				limit = { faith = { faith_has_free_holy_site_action_trigger = yes } }
				multiply = 0
			}
			#ADDED
			else_if = {
				limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
				multiply = 2
			}
		}
	}

	cooldown = { years = holy_site_decision_cooldown_value }

	effect = {
		scope:selected_item.holy_site_county.holder = {
			save_scope_as = county_holder
		}
		root.faith = {
			make_holy_site_eminent = {
				target = scope:selected_item
				actor = root
			}
			consume_free_holy_site_action_effect = yes
		}
		if = {
			limit = {
				exists = scope:county_holder
				NOT = { scope:county_holder = root }
			}
			scope:county_holder = {
				add_opinion = {
					modifier = holy_site_elevated_county_holder_opinion
					target = root
				}
			}
		}
		add_character_modifier = {
			modifier = holy_site_elevated_modifier
			years = 10
		}
	}

	ai_potential = {
		is_independent_ruler = yes
		can_manage_faith_holy_sites_trigger = yes
		faith = {
			list_size:eminent_holy_site < religion.eminent_holy_sites_max
		}
		OR = {
			highest_held_title_tier >= tier_kingdom
			faith.religious_head = this
		}
	}

	ai_will_do = {
		base = 100
	}

	widget = {
		gui = "decision_view_widget_select_holy_site"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT_HOLY_SITE"

		setup_items = {
			scope:actor = {
				faith = {
					every_holy_site = {
						is_eminent = no
						add_to_list = item_list
					}
				}
			}
		}

		default_item = {
			scope:actor = {
				faith = {
					random_holy_site = {
						is_eminent = no
						save_scope_as = selected_item
					}
				}
			}
		}

		ai_select_item = {
			scope:actor = {
				faith = {
					random_holy_site = {
						is_eminent = no
						limit = {
							OR = {
								scope:actor.faith.religious_head =  scope:actor
								holy_site_barony.holder = {
									target_is_same_character_or_above = scope:actor
								}
							}
						}
						save_scope_as = selected_item
					}
				}
			}
		}
	}
}

demote_holy_site_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_demote_holy_site.dds"
	}
	decision_group_type = religious

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_invisible = yes

	is_valid_showing_failures_only = {
		exists = scope:selected_item
	}

	is_valid = {
		save_temporary_scope_as = root
		exists = scope:selected_item

		can_manage_faith_holy_sites_trigger = yes

		scope:selected_item ?= {
			root.faith = {
				is_faith_eminent_holy_site = prev
				save_temporary_scope_as = faith
				custom_tooltip = {
					text = demote_holy_site_decision_above_eminent_holy_sites_min
					list_size:eminent_holy_site > root.faith.religion.eminent_holy_sites_min
				}
			}
			#ADDED - only ones controlled by faith, unless the action is free
			trigger_if = {
				limit = { root.faith = { faith_has_free_holy_site_action_trigger = no } } 
				custom_tooltip = {
					text = elevate_holy_site_decision_county_holder_correct_faith
					holy_site_county.holder.faith = root.faith
				}
				trigger_if = {
					limit = { root.faith = { has_doctrine_parameter = no_head_of_faith } }
					custom_tooltip = {
						text = adopt_holy_site_in_realm
						holy_site_county.holder = { target_is_same_character_or_above = root }
					}
				}
			}
		}
		

		# Show usage of free holy site action, if appropriate
		faith = {
			holy_site_free_action_show_usage_trigger = yes
		}
	}

	cost = {
		piety = {
			value = 1000
			if = {
				limit = { faith = { faith_has_free_holy_site_action_trigger = yes } }
				multiply = 0
			}
			#ADDED
			else_if = {
				limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
				multiply = 2
			}
		}
	}

	cooldown = { years = holy_site_decision_cooldown_value }

	effect = {
		scope:selected_item.holy_site_county.holder = {
			save_scope_as = county_holder
		}
		root.faith = {
			make_holy_site_not_eminent = {
				target = scope:selected_item
				actor = root
			}
			consume_free_holy_site_action_effect = yes
		}
		if = {
			limit = {
				exists = scope:county_holder
				NOT = { scope:county_holder = root }
				scope:county_holder.faith = root.faith
			}
			scope:county_holder = {
				add_opinion = {
					modifier = holy_site_demoted_county_holder_opinion
					target = root
				}
			}
		}
		add_character_modifier = {
			modifier = holy_site_demoted_modifier
			years = 10
		}
		# Demoting a schism-rite holy site feeds Great Schism progress
		hidden_effect = {
			if = {
				limit = {
					root.faith = faith:christian_faith
					pam_schism_county_rite_in_pair_trigger = { COUNTY = scope:selected_item.holy_site_county }
				}
				pam_change_great_schism_progress_effect = {
					VALUE = pam_great_schism_progress_holy_site_value
					TT = pam_great_schism_progress_holy_site_tt
				}
			}
		}
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}

	widget = {
		gui = "decision_view_widget_select_holy_site"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT_HOLY_SITE"

		setup_items = {
			scope:actor = {
				faith = {
					every_eminent_holy_site = {
						add_to_list = item_list
					}
				}
			}
		}

		default_item = {
			scope:actor = {
				faith = {
					random_eminent_holy_site = {
						save_scope_as = selected_item
					}
				}
			}
		}

		ai_select_item = {
			scope:actor = {
				faith = {
					random_eminent_holy_site = {
						save_scope_as = selected_item
					}
				}
			}
		}
	}
}

abandon_holy_site_decision = {
	picture = {
		reference = "gfx/interface/illustrations/event_scenes/temple.dds"
	}
	decision_group_type = religious

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_invisible = yes

	is_valid_showing_failures_only = {
		exists = scope:selected_item
	}

	is_valid = {
		save_temporary_scope_as = root
		exists = scope:selected_item

		can_manage_faith_holy_sites_trigger = yes

		scope:selected_item ?= {
			root.faith = {
				save_temporary_scope_as = faith
				is_faith_holy_site = prev
				NOT = { is_faith_eminent_holy_site = prev }
				custom_tooltip = {
					text = abandon_holy_site_decision_above_holy_sites_min
					list_size:holy_site > root.faith.religion.holy_sites_min
				}
			}
			#ADDED - you either have free actions, control the holy site or have HOF and holy site isnt controlled by faith
			trigger_if = {
				limit = { root.faith = { faith_has_free_holy_site_action_trigger = yes } } 
			}
			trigger_else_if = {
				limit = { root.faith = { has_doctrine_parameter = no_head_of_faith } }
				OR = {
					custom_tooltip = {
						text = abandon_holy_site_decision_county_holder_incorrect_faith
						holy_site_county.holder.faith != root.faith
					}
					custom_tooltip = {
						text = adopt_holy_site_in_realm
						holy_site_county.holder = { target_is_same_character_or_above = root }
					}
				}
			}
			trigger_else = {
			}
		}

		# Show usage of free holy site action, if appropriate
		faith = {
			holy_site_free_action_show_usage_trigger = yes
		}
	}

	cost = {
		piety = {
			value = 1000
			if = {
				limit = { faith = { faith_has_free_holy_site_action_trigger = yes } }
				multiply = 0
			}
			#ADDED
			else_if = {
				limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
				multiply = 2
			}
		}
	}

	cooldown = { years = holy_site_decision_cooldown_value }

	effect = {
		scope:selected_item.holy_site_county.holder = {
			save_scope_as = county_holder
		}
		hidden_effect = {
			# Enshrined relics belong to the site, not the faith, so they go to the county holder even if other faiths keep the site
			scope:selected_item ?= {
				if = {
					limit = { exists = scope:county_holder }
					every_holy_site_enshrined_artifact = {
						set_owner = { target = scope:county_holder }
					}
				}
			}
		}
		root.faith = {
			remove_holy_site = {
				target = scope:selected_item
				actor = root
			}
			consume_free_holy_site_action_effect = yes
		}
		if = {
			limit = {
				exists = scope:county_holder
				NOT = { scope:county_holder = root }
				scope:county_holder.faith = root.faith
			}
			scope:county_holder = {
				add_opinion = {
					modifier = holy_site_demoted_county_holder_opinion
					target = root
				}
			}
		}
		add_character_modifier = {
			modifier = holy_site_demoted_modifier
			years = 10
		}
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}

	widget = {
		gui = "decision_view_widget_select_holy_site"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT_HOLY_SITE"

		setup_items = {

adopt_holy_site_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/pam_decision_mentor_sp.dds"
	}
	decision_group_type = religious

	sort_order = 110

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 24
		hegemony = 24
	}

	is_invisible = yes

	is_valid_showing_failures_only = {
		exists = scope:selected_item
	}

	is_valid = {
		save_temporary_scope_as = root

		is_independent_ruler = yes

		can_manage_faith_holy_sites_trigger = yes

		root.faith = {
			save_temporary_scope_as = faith # used by loc below
			custom_tooltip = {
				text = adopt_holy_site_decision_under_holy_sites_max
				list_size:holy_site < root.faith.religion.holy_sites_max
			}
		}
		exists = scope:selected_item
		scope:selected_item ?= {
			holy_site_county.holder = {
				custom_tooltip = {
					text = adopt_holy_site_in_realm
					target_is_same_character_or_above = root
				}
				faith = root.faith
			}

			custom_tooltip = {
				text = adopt_holy_site_decision_not_already_a_holy_site
				root.faith = {
					NOT = { is_faith_holy_site = prev }
				}
			}
			root.faith = {
				faith_can_use_holy_site_kingdom_trigger = { BARONY = scope:selected_item.holy_site_barony }
			}
			OR = {
				root.rite = { rite_has_parameter = allow_adopt_any_holy_site }
				custom_tooltip = {
					text = adopt_holy_site_decision_shared_or_syncretic_faith
					holy_site_adoptable_by_character_trigger = { CHARACTER = root }
				}
			}
			# This is just to make the player requirements a bit clearer
			custom_tooltip = {
				text = adopt_holy_site_decision_active_site
				any_holy_site_faith = {
					num_county_followers > 0
				}
			}
		}

		# Show usage of free holy site action, if appropriate
		faith = {
			holy_site_free_action_show_usage_trigger = yes
		}
	}

	cost = {
		piety = {
			value = monumental_piety_value
			if = {
				limit = { faith = { faith_has_free_holy_site_action_trigger = yes } }
				multiply = 0
			}
			#ADDED
			else_if = {
				limit = { faith = { has_doctrine_parameter = no_head_of_faith } }
				multiply = 2
			}
		}
	}

	cooldown = { years = holy_site_decision_cooldown_value }

	effect = {
		root.faith = {
			add_holy_site = {
				target = scope:selected_item
				actor = root
			}
			consume_free_holy_site_action_effect = yes
		}
	}

	ai_potential = {
		is_independent_ruler = yes
		can_manage_faith_holy_sites_trigger = yes
		piety >= monumental_piety_value
		faith = {
			list_size:holy_site < {
				value = root.faith.religion.holy_sites_max
				# always leave one slot for a player to adopt/create
				subtract = 1
			}
		}
	}

	ai_will_do = {
		base = 100
	}

	widget = {
		gui = "decision_view_widget_select_holy_site"
		controller = select_scope_object
		show_from_start = yes
		decision_to_second_step_button = "SELECT_HOLY_SITE"

		setup_items = {
			scope:actor = {
				every_realm_county =  {
					every_county_holy_site = {
						limit = {
							scope:actor.faith = {
								NOT = { is_faith_holy_site = prev }
							}
						}
						add_to_list = item_list
					}
				}
			}
		}

		is_item_valid = {
			scope:actor.faith = {
				NOT = { is_faith_holy_site = root }
			}
			holy_site_county.holder = {
				faith = scope:actor.faith
			}
			custom_tooltip = {
				text = adopt_holy_site_decision_active_site
				any_holy_site_faith = {
					num_county_followers > 0
				}
			}
			scope:actor.faith = {
				faith_holy_site_kingdom_unused_trigger = { BARONY = root.holy_site_barony }
			}
		}

		ai_item_will_do = {
			value = 0
			if = {
				limit = {
					OR = {
						scope:actor.rite = { rite_has_parameter = allow_adopt_any_holy_site }
						holy_site_adoptable_by_character_trigger = { CHARACTER = scope:actor }
					}
					scope:actor.faith = {
						faith_holy_site_kingdom_unused_trigger = { BARONY = root.holy_site_barony }
					}
				}
				add = 100
			}
		}
	}
}
