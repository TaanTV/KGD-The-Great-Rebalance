adventurers_scout_for_talent_option = {
	is_shown = {
   		has_government = landless_adventurer_government
		NOT = { has_perk = toe_the_line_perk }
	}

	is_valid = {
		NOT = { has_perk = toe_the_line_perk }
		has_government = landless_adventurer_government
	}
	
	travel_modifier = {
		travel_speed = -5
	}
	
	on_applied_effect = {
		custom_tooltip = adventurers_scout_for_talent_option.tt
	}
	
	ai_will_do = {
		value = 0		#prev 100
		#ADDED
		if = {
			limit = { travel_plan_owner.court_positions_currently_available > 0 }
			value = 10000
		}
	}
}
