
# called for armies whenever they enter a province
# root is the owner of the army
# scope:army is the army

on_army_enter_province = {
	#ADDED THIS IS AN OVERWRITE OF THE EFFECT
	effect = {
		save_scope_as = army_owner
		# Commander trait leveling
		scope:army.army_commander ?= {
			if = {
				limit = {
					has_trait = logistician
					#ADDED - no. Moved to general XP gain.
					always = no
				}
				add_trait_xp = {
					trait = logistician
					value = {
						value = 0.2
						add = {
							value = martial
							divide = 20
						}
					}
				}
			}
			if = {
				limit = { 
					has_trait = organizer 
					#ADDED - no. Moved to general XP gain.
					always = no
				}
				add_trait_xp = {
					trait = organizer
					value = {
						value = 0.2
						add = {
							value = martial
							divide = 20
						}
					}
				}
			}
		}
		# Army movement damages Trade
		if = {
			limit = {
				# Prevent same army owner affecting many markets in quick succession
				NOT = { exists = var:army_movement_trade_market_cooldown }
				scope:army ?= {
					# Prevent same army affecting many markets in quick succession
					NOT = { exists = var:army_movement_trade_market_cooldown }
					# Ensure tiny armies are not messing up trade
					army_size > 250
					location = {
						barony ?= {
							# Prevent same market being affected too often
							NOT = { exists = trade_market.var:army_movement_trade_market_cooldown }
							# Ignore occupied land
							barony_controller != scope:army_owner
						}
						# Prevent damaging own or allied markets directly (overlapping is fine)
						county.holder ?= {
							save_temporary_scope_as = county_holder
							OR = {
								root = { is_causing_raid_hostility_towards = scope:county_holder }
								is_at_war_with = root
							}
						}
					}
				}
			}
			random = {
				chance = 1
				modifier = {
					add = {
						value = scope:army.army_size
						multiply = 0.005
						max = 10
					}
				}
				modifier = {
					factor = 0.5
					scope:army.location.barony.trade_market = { has_trade_market_modifier = gdp_market_army_movement_capacity_modifier }
				}
				scope:army = {
					location = {
						save_scope_as = army_province
						barony.trade_market = {
							save_scope_as = trade_market
							random_list = {
								1 = {
									trigger = {
										any_trade_good_with_surplus = {}
									}
									random_trade_good_with_surplus = { save_scope_as = trade_good }
								}
								1 = {}
							}
							if = { # Increase price of a locally traded good
								limit = { exists = scope:trade_good }
								add_trade_good_price_modifier = {
									type = buy_add
									trade_good_type = scope:trade_good
									value = 0.01
									years = 5
								}
								every_character_in_market = {
									limit = { government_allows = profit_trade_routes }
									send_interface_message = {
										type = msg_army_affects_trade
										title = msg_army_movement_affects_trade_title
										left_icon = root
										right_icon = scope:army.location.barony
										desc = msg_army_movement_affects_trade_desc
										show_as_tooltip = {
											scope:trade_market = {
												add_trade_good_price_modifier = {
													type = buy_add
													trade_good_type = scope:trade_good
													value = 0.01
													years = 5
												}
											}
										}
									}
								}
							}
							else = { # Disrupt local prices across the market
								add_trade_market_modifier = {
									modifier = gdp_market_army_movement_capacity_modifier
									years = 5
								}
								every_character_in_market = {
									limit = { government_allows = profit_trade_routes }
									send_interface_message = {
										type = msg_army_affects_trade
										title = msg_army_movement_affects_trade_title
										left_icon = root
										right_icon = scope:army.location.barony
										desc = msg_army_movement_affects_trade_desc
										show_as_tooltip = {
											scope:trade_market = {
												add_trade_market_modifier = {
													modifier = gdp_market_army_movement_capacity_modifier
													years = 5
												}
											}
										}
									}
								}
							}
							# Prevent same trade market being affected too often
							set_variable = {
								name = army_movement_trade_market_cooldown
								months = 6
							}
						}
						# Prevent same army affecting trade markets too often
						set_variable = {
							name = army_movement_trade_market_cooldown
							months = 6
						}
					}
				}
				# Prevent same army owner affecting trade markets too often
				set_variable = {
					name = army_movement_trade_market_cooldown
					months = 6
				}
			}
		}
	}
}

# Called when a raid action starts
# root is the raid army
# scope:raider is the person owning the raid army
# scope:barony is the barony title that got raided
# scope:county is the county title for the barony
on_raid_action_start = {
	effect = { # The target isn't too keen on the raider...
		scope:barony.holder = {
			if = {
				limit = {
					#ADDED - nah
					always = no
					has_opinion_modifier = {
						target = scope:raider
						modifier = raided_me_opinion
					}
				}
				remove_opinion = { # We reapply it so that the counter is reset
					target = scope:raider
					modifier = raided_me_opinion
				}
			}
			add_opinion = {
				target = scope:raider
				modifier = raided_me_opinion
			}
			every_liege_or_above = {
				if = {
					limit = {
						#ADDED - nah
						always = no
						has_opinion_modifier = {
							target = scope:raider
							modifier = raided_me_opinion
						}
					}
					remove_opinion = { # We reapply it so that the counter is reset
						target = scope:raider
						modifier = raided_me_opinion
					}
				}
				add_opinion = {
					target = scope:raider
					modifier = raided_me_opinion
				}
			}
		}
	}
}

# Called when a raid action finishes
# root is the raid army
# scope:raider is the person owning the raid army
# scope:barony is the barony title that got raided
# scope:county is the county title for the barony
on_raid_action_completion = {
	effect = {
		scope:raider = {
			send_interface_message = {
				type = msg_raid_completed
				left_icon = scope:county.holder
				right_icon = scope:barony
				desc = msg_raid_completed_desc
				show_as_tooltip = {
					scope:barony.trade_market = {
						add_trade_market_modifier = {
							modifier = gdp_market_recently_raided_capacity_modifier
							years = 5
						}
					}
				}
			}
			if = {
				limit = { 
					scope:county = {
						any_county_province = { has_special_district = yes }
					}
					NOT = { has_character_flag = raider_legend }
					has_dlc_feature = legends
					NOT = { has_game_rule = historical_legends_only }
				}
				random = {
					chance = legend_seed_chance_low
					add_character_flag = raider_legend
					create_legend_seed = {
						type = heroic
						quality = famed
						chronicle = legendary_battle
						properties = {
							winner = scope:raider
							loser = scope:county.holder
							location = scope:barony.title_province
						}
					}
				}
			}
			house ?= {
				if = {
					limit = {
						exists = scope:county.holder.house
					}
					#ADDED - a bit scuffed tabulation for better autocomparing
					random = {
						chance = 50
						modifier = {
							NOT = { has_house_relation_with = scope:county.holder.house }
							add = -17
						}
						modifier = {
							scope:barony.holder != scope:county.holder
							factor = 0.5
						}
					change_house_relation_effect = {
						HOUSE = scope:county.holder.house
						VALUE = house_relation_damage_minor_value
						REASON = raid
						CHAR = scope:raider
						TARGET_CHAR = scope:county.holder
						TITLE = scope:dummy_gender
					}
					}
				}
			}
		}
		scope:barony = {
			title_province = {
				add_province_modifier = {
					modifier = recently_looted_modifier
					years = 5
				}
			}
			trade_market = {
				add_trade_market_modifier = {
					modifier = gdp_market_recently_raided_capacity_modifier
					years = 5
				}
			}
			county = {
				if = {
					limit = {
						holder = {
							NOT = { has_perk = a_mans_home_perk }
						}
					}
					change_county_control = raid_county_control_loss_value
					random_list = {
						25 = {
							change_prosperity_progress_with_overflow = -6	#prev -2
						}
						25 = {
							change_prosperity_progress_with_overflow = -9	#prev -3
						}
						25 = {
							change_prosperity_progress_with_overflow = -12	#prev -4
						}
						25 = {
							change_prosperity_progress_with_overflow = -15	#prev -5
						}
					}
				}
				# Silk Road
				tgp_silk_road_raid_effect = yes
			}
		}
		scope:county.holder = {
			add_to_list = raid_targets
			every_liege_or_above = { add_to_list = raid_targets }
		}
		every_in_list = {
			list = raid_targets
			limit = {
				NOT = { this = scope:raider }
			}
			send_interface_message = {
				type = msg_raid_lost
				left_icon = scope:raider
				right_icon = scope:barony
				desc = msg_raid_lost_desc
				show_as_tooltip = {
					scope:barony.trade_market = {
						add_trade_market_modifier = {
							modifier = gdp_market_recently_raided_capacity_modifier
							years = 5
						}
					}
				}
			}
		}

		## Struggles
		if = {
			limit = {
				scope:raider = {
					any_character_struggle = {
						activate_struggle_catalyst_secondary_character_involvement_involved_trigger = {
							CATALYST = catalyst_raided_involved
							CHAR = scope:county.holder
						}
					}
				}
				scope:county.holder = { # Theoretical edge case where X is from one struggle, Y from another one and they trigger catalysts. But unlikely to be an issue.
					any_character_struggle = {
						involvement = involved
						phase_has_catalyst = catalyst_raided_involved
					}
				}
			}
			scope:raider = {
				every_character_struggle = {
					limit = {
						activate_struggle_catalyst_secondary_character_involvement_involved_trigger = {
							CATALYST = catalyst_raided_involved
							CHAR = scope:county.holder
						}
					}
					activate_struggle_catalyst = {
						catalyst = catalyst_raided_involved
						character = scope:raider
					}
					log_debug_variable_for_persian_struggle_effect = { VAR = concession_catalyst_raided_involved }
				}
			}
		}

		if = { # PAM - Catalysts
			limit = {
				pam_neighboring_unreformed_pagan_raid_you_catalyst_trigger = yes
			}
			situation:the_christian_church = {
				trigger_situation_catalyst = catalyst_neighboring_unreformed_pagan_raid_you
			}
		}
		
		# Commander trait leveling
		root.army_commander ?= {
			if = {
				limit = {
					has_trait = reaver
					is_commanding_army = yes
				}
				add_trait_xp = {
					trait = reaver
					value = {
						value = 0.5
						add = {
							value = martial
							divide = 10
						}
					}
				}
			}
		}

		tgp_treasury_raided_effect = yes
	}
}
