# A title is destroyed
# root = the holder before destruction
# scope:landed_title = the title that is being destroyed
# Does *not* provide scope:transfer_type
on_title_destroyed = {
	on_actions = {
		BKT_on_empire_destroyed
	}
}

BKT_on_empire_destroyed = {
	trigger = {
		scope:landed_title.tier >= tier_empire
	}
	effect = {
		scope:landed_title = { 
			remove_variable = year_of_last_serious_defensive_war 
			#remove_variable = number_of_successions		#No idea how to track that one properly...
		}
	}
}

# A title is transferred to a new character
# root = the new holder
# scope:title = the title that changes hands
# scope:previous_holder = previous holder. Might be dead
#Sadly there is no separate on_title_created
on_title_gain = {
	on_actions = {
		BKT_on_empire_created
		BKT_on_barony_gained
	}
}

BKT_on_empire_created = {
	trigger = {
		scope:transfer_type ?= flag:created
		scope:title.tier >= tier_empire
		#Those have their own thingies
		save_temporary_scope_as = temp_ruler_scope
		NOT = {
			any_character_situation = { 
				situation_type = dynastic_cycle 
				any_participant_group = {
					NOT = { participant_group_type = other_rulers }
					participant_group_has_character = scope:temp_ruler_scope
				}
			} 
		}
	}
	effect = {
		scope:title = { save_scope_as = landed_title }
		get_or_set_year_of_last_serious_defensive_war = yes
	}
}

# A title change makes the character rank up in the landed tiers (eg Duke -> King)
# root = character ranking up
# scope:title = old primary title
on_rank_up = { # Will not fire during history execution or for dying characters
	on_actions = {
		BKT_on_landed
		delay = { days = 1 }
		BKT_on_landed_treasury_fix
	}
}

BKT_on_landed = {
	trigger = {
		NOT = { exists = scope:title }
		is_landed = yes
	}
	effect = {
		if = {
			limit = { 
				top_liege != this
				highest_held_title_tier > tier_barony
				NOT = { government_has_flag = government_is_tribal }
				liege = { government_has_flag = government_is_tribal }
				OR = {
					culture ?= liege.culture
					culture.culture_head ?= { government_has_flag = government_is_tribal }
				}
				primary_title ?= { recent_history = { type = granted days = 1 } }
			}
			if = {
				limit = { liege = { has_government = tribal_government } }
				change_government = tribal_government
			}
			else_if = {
				limit = { liege = { has_government = wanua_government } }
				change_government = wanua_government
			}
		}
	}
}


# A vassal changes their liege
# root = vassal changing their status
# scope:old_liege = the previous liege of the vassal
# It is also fired for rulers gaining independence, those can be identified by checking current liege
on_vassal_change = {
	on_actions = {
		BKT_vassal_change_tyrannical_changes_maintenance
	}
}

BKT_vassal_change_tyrannical_changes_maintenance = {
	trigger = {
		vassal_contract_has_modifiable_obligations = yes
		has_variable = had_their_contract_tyranically_changed
	}
	effect = {
		save_temporary_scope_as = root_scope
		if = {
			limit = { top_liege = this }
			remove_variable = had_their_contract_tyranically_changed
			set_vassal_contract_modification_blocked = no
			scope:old_liege ?= {
				if = {
					limit = { 
						is_alive = yes 
						has_variable_list = had_their_contract_tyranically_changed_list
						any_in_list = {
							variable = had_their_contract_tyranically_changed_list
							this = scope:root_scope
						} 
					}
					remove_list_variable = {
						name = had_their_contract_tyranically_changed_list
						target = scope:root_scope
					}
				}
			}
		}
		else_if = {
			limit = { 
				subject_contract_is_blocked_from_modification = yes
				var:had_their_contract_tyranically_changed = scope:old_liege
			}
			set_vassal_contract_modification_blocked = no
			#add toasts and stuff
			if = {
				limit = { scope:old_liege = { is_alive = yes } }
				send_interface_message = {
					type = send_interface_message
					title = contract_unlocked_by_liege_change_death_title
					#desc = negative_piety_convertion_desc
					right_icon = root
					custom_tooltip = contract_unlocked_by_liege_change_tt
					show_as_tooltip = { set_vassal_contract_modification_blocked = no }
				}
			}
			else = {
				send_interface_message = {
					type = send_interface_message
					title = contract_unlocked_by_liege_change_death_title
					#desc = negative_piety_convertion_desc
					right_icon = root
					custom_tooltip = contract_unlocked_by_liege_death_tt
					show_as_tooltip = { set_vassal_contract_modification_blocked = no }
				}
			}
		}
	}
}

BKT_on_barony_gained = {
	trigger = {
		OR = {
			scope:transfer_type = flag:granted
			scope:transfer_type = flag:appointment
		}
		scope:title.tier = tier_barony
	}
	effect = { 
		set_variable = {
			name = got_barony_granted
			days = 2
		}
	}
}

BKT_on_landed_treasury_fix = {
	trigger = {
		NOT = { exists = scope:title }
		government_allows = treasury
		#gold = treasury
		gold > 0
		OR = {
			AND = {
				highest_held_title_tier = tier_barony
				has_variable = got_barony_granted
			}
			primary_title ?= {
				recent_history = { type = appointment days = 1 }
			}
		}
	}
	effect = {
		remove_treasury = gold
	}
}
