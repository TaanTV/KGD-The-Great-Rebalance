# character having its birthday in root scope
# tested after age has increased
on_birthday = {
	on_actions = {
		bkt_on_child_birthday_traits
	}
}

bkt_on_child_birthday_traits = {
	trigger = {
		is_adult = no
	}
	effect = {
		if = {
			limit = { age >= 14 }
			remove_character_modifier = just_a_child_character_modifier
			remove_character_modifier = still_a_child_character_modifier
		}
		else_if = {
			limit = { age >= 7 }
			if = {
				limit = { NOT = { has_character_modifier = still_a_child_character_modifier } }
				add_character_modifier = still_a_child_character_modifier
			}
			remove_character_modifier = just_a_child_character_modifier
		}
		else = {
			if = {
				limit = { NOT = { has_character_modifier = just_a_child_character_modifier } }
				add_character_modifier = just_a_child_character_modifier
			}
		}
	}
	on_actions = {
		delay = { days = { 30 150 } }
		bkt_on_child_birthday_tenets
		delay = { days = { 180 330 } }
		bkt_on_child_birthday_tenets
	}
}

bkt_on_child_birthday_tenets = {
	trigger = { age > 4 is_playable_character = yes }
	effect = {
		#keep in mind this happens like 20 times total
		random = {
			chance = {
				add = 2
				add = learning
				max = 10		#sorry 40 learning babies, not this time
			}
			
			save_scope_as = child_scope
			if = {
				limit = {
					any_relation = {
						type = guardian
						is_available = yes
					}		
				}
				random_relation = {
					type = guardian
					save_scope_as = guardian
				}
			}
			else_if = {
				limit = {
					exists = court_owner.court_position:court_tutor_court_position
				}
				court_owner.court_position:court_tutor_court_position ?= {
					save_scope_as = guardian
				}
			}
			else_if = {
				limit = {
					exists = court_owner.court_position:court_guru_court_position
				}
				court_owner.court_position:court_guru_court_position ?= {
					save_scope_as = guardian
				}
			}
			else_if = {
				limit = { exists = host }
				random = {
					chance = 50
					host = { 
						save_scope_as = guardian
					}
				}
			}
			
			if = {
				limit = {
					exists = scope:guardian
					scope:guardian ?= {
						any_character_tenet = {
							knowledge_filter = known
							NOR = {
								scope:child_scope = { knows_tenet = prev }
								scope:guardian.rite ?= {
									has_tenet_status = {
										tenet = prev
										status = prohibited
									}
								}
								AND = {
									tenet_has_virtue = lustful
									scope:child_scope.age < 13
								}
							}
						}
					}
				}
				scope:guardian = {
					random_character_tenet = {
						knowledge_filter = known
						limit = {
							NOR = {
								scope:child_scope = { knows_tenet = prev }
								scope:guardian.rite ?= {
									has_tenet_status = {
										tenet = prev
										status = prohibited
									}
								}
								AND = {
									tenet_has_virtue = lustful
									scope:child_scope.age < 13
								}
							}
						}
						save_scope_as = passive_learn_tenet
					}
					save_scope_as = passive_learn_source
				}
				add_known_tenet = scope:passive_learn_tenet
				send_interface_message = {
					type = msg_passive_learned_tenet
					desc = msg_passive_learned_tenet_desc
					left_icon = scope:passive_learn_source
					right_icon = scope:passive_learn_tenet
				}
			}
		}
	}
}
