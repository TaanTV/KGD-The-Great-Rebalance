on_game_start = {
	on_actions = {
		BKT_loaded_on_action
		BKT_error_supress_on_action
		BKT_admin_ghana_fix
		BKT_on_start_female_pool
		BKT_on_start_misc
	}
}

BKT_loaded_on_action = {
	effect = {
		set_global_variable = KGD_the_great_rebalance_is_loaded
		set_global_variable = got_war_dec_laws_fixed
	}
}

BKT_error_supress_on_action = {
	trigger = {
		always = no 
	}
	effect = {
		if = {
			limit = { 
				has_global_variable = KGD_the_great_rebalance_is_loaded
				has_global_variable = KGD_schizo_balance_is_loaded
			}
			set_global_variable = KGD_schizo_balance_is_loaded
		}
		random_county = {
			title_province ?= {
				if = { limit = { has_province_modifier = somewhat_recently_looted_modifier } }
			}
			holder ?= {
				if = { 
					limit = { 
						has_character_modifier = dummy_mp_rules_description_modifier 
						has_character_modifier = support_religious_art_bkt_county_modifier 
						has_character_modifier = artifact_domain_limit_1_modifier 	#not sure if i even need to keep it or not
						has_variable = currently_dying
					} 
				}
				set_variable = got_secret_faith_through_mental_break_event
				set_variable = heresy_power
				add_to_variable_list = {
					name = entourage_members_to_kill
					target = this
				}
			}
		}
	}
}

BKT_admin_ghana_fix = {
	trigger = {
		has_game_rule = historicity_extra_admin_ghana_historic_always
		current_date < 868.1.1
		exists = title:k_ghana.holder
	}
	effect = {
		title:k_ghana = {
			holder = {
				every_sub_realm_county = {
					limit = {
						title_province = { has_holding_type = tribal_holding }
					}
					title_province = { 
						set_holding_type = castle_holding 
						generate_building = yes
						random = {
							chance = {
								value = 25
								if = {
									limit = { duchy.title_capital_county.title_province = this }
									value = 50
								}
							}
							generate_building = yes
						}
					}
					random_county_province = {
						limit = { 
							has_holding = no 
							NOT = { barony_cannot_construct_holding = yes }
						}
						random_list = {
							10 = {
								set_holding_type = city_holding
								random = {
									chance = 50
									generate_building = yes
								}
							}
							10 = {
								set_holding_type = church_holding
								random = {
									chance = 50
									generate_building = yes
								}
							}
							60 = { }	#to retain the original 25% of this happening at all
						}
					}
				}
			}
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
			}
			every_in_de_jure_hierarchy = {
				limit = { 
					tier = tier_duchy 
					title:k_ghana.holder = { completely_controls = prev }
				}
				change_title_holder = {
					holder = title:k_ghana.holder
					change = scope:change
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
	}
}

		BKT_initial_opinion_boost
	}

		game_start_date = 867.1.1

BKT_on_start_female_pool = {
	effect = {
		every_duchy = {
			title_capital_county ?= {
				random = {
					chance = 33
					
					save_temporary_scope_as = title_of_origin
					create_character = {
						age = { 16 27 }
						gender = female
						dynasty = generate
						location = scope:title_of_origin.title_province
						faith = scope:title_of_origin.faith
						rite = scope:title_of_origin.rite
						culture = scope:title_of_origin.culture
						random_traits = yes
						#save_scope_as = bookmaker
						after_creation = {
							#debug_log = "generated someone at start"
							#debug_log_scopes = yes
							random = {
								chance = 50
								dynasty ?= { add_dynasty_prestige_level = 1 }
							}
						}
					}
				}
			}
		}
	}
}

BKT_initial_opinion_boost = {
	effect = {
		every_ruler = {
			#moved this here to not iterate every_ruler too much
			if = {
				limit = { prestige < 0 }
				add_prestige_no_experience = {
					add = 150	#start with at least 150
					multiply = { value = highest_held_title_tier add = -1 min = 1 }
					subtract = prestige
				}
			}
			#Help nomads out a bit
			if = {
				limit = { has_government = nomad_government }
				domicile ?= {
					change_herd = {
						value = owner.highest_held_title_tier
						add = 1
						multiply = owner.highest_held_title_tier
						multiply = owner.highest_held_title_tier
						multiply = 85
						multiply = { 0.8 1.25 }
					}
				}
			}
			#Make karlings male_only
			#if = {
			#	limit = { 
			#		dynasty:25061 ?= dynasty 
			#	}
			#	add_realm_law_skip_effects = male_only_law
			#	if = {
			#		limit = { any_held_title ?= { title_tier >= duchy has_title_law = male_only_law } }
			#		every_held_title = { title_tier >= duchy remove_title_law = male_only_law }
			#	}
			#}
			every_courtier = {
				limit = { NOT = { has_relation_rival = prev } }
				add_opinion = {
					target = prev
					modifier = respect_opinion
					opinion = 30
				}
			}
			#Im pretty sure this should be more performant than iterating on every vassal
			if = {
				limit = { top_liege != this NOT = { has_relation_rival = liege } }
				add_opinion = {
					target = liege
					modifier = respect_opinion
					opinion = 30
				}
			}	
		}
	}
