# Whether a character can create a new dynamic holy site.
# root                 = the character creating the holy site
# scope:title          = the barony the holy site would be placed in
# scope:holy_site_type = the holy site type being created
# scope:faith          = the faith the holy site is created for
can_create_dynamic_holy_site = {
	can_manage_faith_holy_sites_trigger = yes

	# This is a command, not a decision, so the variable cooldown is set in on_character_holy_site_created
	custom_description = {
		text = holy_site_creation_on_cooldown_tt
		NOT = { has_variable = holy_site_creation_cooldown }
	}

	scope:faith = {
		faith_can_use_holy_site_kingdom_trigger = { BARONY = scope:title }
		
		#ADDED - does this localize?
		scope:title.duchy = { save_temporary_scope_as = duchy_title_scope }
		custom_tooltip = {
			text = faith_already_has_holy_site_in_duchy
			NOT = {
				any_holy_site_barony = {
					duchy ?= scope:title.duchy
				}
			}
		}
	}
}
