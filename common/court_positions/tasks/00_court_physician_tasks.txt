physician_treat_epidemic = {
	court_position_types = { court_physician_court_position }

	# Monthly cost for the task
	cost = {
		round = no
		gold = {
			value = 0
			if = {
				limit = {
					scope:liege = {
						has_treasury = no
					}
				}
				add = {
					value = monthly_court_position_task_cost
					desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
					format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
				}
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					scope:liege = {
						has_treasury = yes
					}
				}
				add = {
					value = monthly_court_position_task_cost
					desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
					format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
				}
			}
		}
	}

	is_shown = {
		#is_landed_or_landless_administrative_or_theocratic = yes
		#ADDED
		is_landed = yes
	}

	employee_modifier = {
		monthly_prestige = minor_court_position_task_prestige_salary
	}

	scaling_employer_modifiers = {
		terrible = {
		 	epidemic_resistance = 2
		}
		poor = {
		 	epidemic_resistance = 5
		}
		average = {
		 	epidemic_resistance = 7
		}
		good = {
		 	epidemic_resistance = 10
		}
		excellent = {
		 	epidemic_resistance = 15
		}
	}

	ai_will_do = {
		value = 0	#prev 30
		if = {
			limit = {
				any_held_county = {			#prev any_sub_realm_county
					any_county_province_epidemic = {
					}
				}
				NOT = { has_trait = avaricious }
			}
			#multiply = 1.5
			#ADDED
			value = 1000
		}
		if = {
			limit = {
				#realm_has_any_epidemic = { SIZE = 1 INTENSITY = major }
				#ADDED
				any_held_county = { any_county_province_epidemic = { OR = { outbreak_intensity = major outbreak_intensity = apocalyptic } } }
			}
			multiply = 2
		}
		if = {
			limit = {
				capital_county ?= {
					any_county_province_epidemic = {
					}
				}
			}
			multiply = 2
		}
		if = {
			limit = {
				has_treasury = no
				monthly_character_income < monthly_court_position_task_cost_double
				NOT = {
					capital_county ?= {
						any_county_province_epidemic = {
							outbreak_intensity >= major
						}
					}
				}
			}
			add = -1000
		}
		else_if = {
			limit = {
				has_treasury = yes
				monthly_character_treasury_income < monthly_court_position_task_cost_double
				NOT = {
					capital_county ?= {
						any_county_province_epidemic = {
							outbreak_intensity >= major
						}
					}
				}
			}
			add = -1000
		}
	}
}

physician_improve_self = {
	court_position_types = { court_physician_court_position }

	# Monthly cost for the task
	cost = {
		round = no
		prestige = {
			add = {
				value = monthly_court_position_task_cost
				desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
				format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
			}
		}
	}

	employee_modifier = {
		lifestyle_physician_xp_gain_mult = 0.25
	}

	on_start = {
		custom_tooltip = physician_improve_self_fallback_tt
	}

	on_monthly = {
		random = {
			chance = {
				value = 5				#prev 3
				add = {
					value = "aptitude(court_physician_court_position)"
					multiply = -1		#prev 0.5
				}
				#ADDED - traits!
				if = {
					limit = {	#diligent work HARD and cover all the penalties
						has_trait = diligent
					}
					add = 1
				}
				else_if = {
					limit = {	 #lazy and arrogant ones dont improve much
						OR = {
							has_trait = lazy
							has_trait = arrogant
						}
					}
					add = -2
				}
				else_if = {
					limit = {	#fickle and deceitful dont improve much
						OR = {
							has_trait = fickle
							has_trait = deceitful
						}
					}
					add = -1
				}
				if = {
					limit = { is_ruler = yes }
					multiply = 0.5
				}
				if = {
					limit = { OR = { has_trait = blind has_trait = maimed } }
					multiply = 0.3
				}
				if = {
					limit = { has_trait = infirm }
					max = 1
					multiply = 0.1
				}
			}
			if = {
				limit = {
					NOT = { has_trait = lifestyle_physician }
				}
				add_trait = lifestyle_physician
			}
			else = {
				random_list = {
					30 = {
						add_learning_skill = 1
					}
					70 = {
						trigger = {
							has_trait_xp = {
								trait = lifestyle_physician
								value <= 100
							}
						}
						add_trait_xp = {
							trait = lifestyle_physician
							value = 3
						}
					}
				}
			}
		}
	}

	ai_will_do = {
		value = 50 # Always good
		#ADDED
		if = {
			limit = { scope:employee = { has_trait = infirm } }
			value = 0
		}
		else_if = {
			limit = { 
				scope:employee = { 
					"aptitude(court_physician_court_position)" < 3 #less than "good"
					NOR = { 
						has_trait = blind 
						has_trait = maimed 
					} 
				}
			}
			value = 1000
		}
		else_if = {
			limit = { 
				scope:employee = { 
					"aptitude(court_physician_court_position)" > 3 #as in, maxxed
				}
			}
			value = 0
		}
		if = {
			limit = {
				#has_treasury = no
				#monthly_character_income < monthly_court_position_task_cost_double
				#ADDED
				prestige < major_prestige_value
				"scope:employee.aptitude(court_physician_court_position)" > 0	#keep improving them into prestige debt if they're terrible
			}
			add = -1000
		}
		else_if = {
			limit = {
				#ADDED - what?
				always = no
				has_treasury = yes
				monthly_character_treasury_income < monthly_court_position_task_cost_double
			}
			add = -1000
		}
	}
}
